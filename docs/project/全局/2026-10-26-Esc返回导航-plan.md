```go
// pagerKeyString raw 按键 → pager 命令串；ok=false 表示该键不参与分发（方向键等忽略）
func pagerKeyString(k termkey.Key) (string, bool) {
	switch k.Kind {
	case termkey.KindRune:
		return strings.ToLower(string(k.R)), true
	case termkey.KindEnter:
		return "", true
	case termkey.KindEsc:
		return "esc", true
	case termkey.KindCtrlC:
		return "ctrl+c", true
	case termkey.KindCtrlD:
		return "ctrl+d", true
	}
	return "", false
}

// pagerKeySource 统一取键源：raw 单键（sess 非 nil）或降级行读取；next 阻塞取一条可分发命令
type pagerKeySource struct {
	sess *keySession
	br   *bufio.Reader
}

func (p *pagerKeySource) next() (string, bool) {
	if p.sess != nil {
		for {
			k, ok := p.sess.kr.Read()
			if !ok {
				return "", false
			}
			if s, dispatch := pagerKeyString(k); dispatch {
				return s, true
			}
		}
	}
	line, err := p.br.ReadString('\n')
	if err != nil {
		return "", false
	}
	return strings.TrimSpace(line), true
}

// newPagerKeySource raw 可用走单键会话，否则降级行读取；调用方 defer sess.close() 还原终端
func newPagerKeySource() (*pagerKeySource, *keySession) {
	sess := &keySession{}
	if sess.resume() {
		return &pagerKeySource{sess: sess}, sess
	}
	return &pagerKeySource{br: bufio.NewReader(os.Stdin)}, sess
}
```

同时在 `view.go` import 块加入 `"github.com/bugzi/gs/internal/termkey"`。

```go
func runPager(ctx *Ctx, render func() string, editPath func() (string, bool), keep bool) error {
	if !keep {
		if path, ok := editPath(); ok {
			if data, err := os.ReadFile(path); err == nil && !edit.ShouldCliEdit(data) {
				fmt.Fprintf(ctx.Out, "长文件：正在调起外部编辑器打开 %s ...（关闭文件后回到 gs）\n", ui.Path(path))
				return editOpen(ctx, path)
			}
		}
	}
	src, sess := newPagerKeySource()
	defer sess.close()
	for {
		body := render()
		fmt.Fprint(ctx.Out, "\n")
		if h, ok := ui.PageSize(); ok {
			ui.PagedPrint(ctx.Out, src.next, body, h)
		} else {
			fmt.Fprintf(ctx.Out, "%s\n", body)
		}
		fmt.Fprint(ctx.Out, ui.Dim("[e]编辑 [r]刷新 [q/Esc]退出 > "))
		line, ok := src.next()
		if !ok {
			return nil // Ctrl-D / 读取失败结束查看
		}
		switch line {
		case "q", "", "esc", "ctrl+c", "ctrl+d":
			return nil
		case "r":
			continue
		case "e":
			path, ok := editPath()
			if !ok {
				fmt.Fprintln(ctx.Out, ui.Dim("本视图没有可编辑的文件"))
				continue
			}
			if src.sess != nil {
				sess.close() // 编辑器需要普通终端，先退出 raw
			}
			fmt.Fprintf(ctx.Out, "正在打开 %s ...（编辑器退出后自动刷新）\n", ui.Path(path))
			if err := editOpen(ctx, path); err != nil {
				return err
			}
			if src.sess != nil && !sess.resume() {
				return nil
			}
		}
	}
}
```

行为说明（对照旧实现）：降级路径逐字保留（`br.ReadString('\n')` 移入 `pagerKeySource.next`）；raw 路径新增 `esc`/`ctrl+c`/`ctrl+d` 退出与 `q/Esc` 提示文案；`""`（回车）在翻页阶段=下一页、在命令提示处=退出，与旧行为一致。

```go
// runShowPager 逐节浏览；o 打开当前节文件，s 沉淀，q/Esc 退出
func runShowPager(ctx *Ctx, v *showView) error {
	cur := 0
	src, sess := newPagerKeySource()
	defer sess.close()
	for {
		sec := v.Sections[cur]
		body := sec.Body
		if strings.TrimSpace(body) == "" {
			body = "（无相关内容）"
		} else {
			body = ui.RenderMD(body)
		}
		fmt.Fprintf(ctx.Out, "\n═══ [%d/5] %s ═══\n", cur+1, sec.Title)
		if h, ok := ui.PageSize(); ok {
			ui.PagedPrint(ctx.Out, src.next, body, h)
		} else {
			fmt.Fprintf(ctx.Out, "%s\n", body)
		}
		fmt.Fprint(ctx.Out, ui.Dim("[n]下一节 [p]上一节 [o]打开文件 [s]沉淀为索引 [q/Esc]退出 > "))
		line, ok := src.next()
		if !ok {
			return nil
		}
		switch line {
		case "n", "":
			if cur < 4 {
				cur++
			}
		case "p":
			if cur > 0 {
				cur--
			}
		case "q", "esc", "ctrl+c", "ctrl+d":
			return nil
		case "o":
			if len(sec.Files) == 0 {
				fmt.Fprintln(ctx.Out, ui.Dim("本节没有可打开的文件"))
				continue
			}
			opts := make([]menuOption, 0, len(sec.Files))
			for _, f := range sec.Files {
				opts = append(opts, option(f.Path+hintSuffix(f.Hint), f.Path))
			}
			var pick string
			if src.sess != nil {
				sess.close() // 子交互（选择菜单/编辑器）前让出终端
			}
			if err := selectFrom(ctx, "打开哪个文件", opts, &pick); err != nil {
				return err
			}
			if err := editOpen(ctx, filepath.Join(ctx.WS.Root, filepath.FromSlash(pick))); err != nil {
				return err
			}
			if src.sess != nil && !sess.resume() {
				return nil
			}
		case "s":
			if src.sess != nil {
				sess.close() // 沉淀输出在普通终端打印，避免 raw 帧残留
			}
			if v.Indexed {
				fmt.Fprintln(ctx.Out, ui.Dim("该主题已在索引中，/show edit 可维护"))
			} else {
				if err := showSave(ctx.WS, v); err != nil {
					return err
				}
				fmt.Fprintf(ctx.Out, "已沉淀 %q 到 %s\n", v.Topic, ui.Path(".gs/topics.toml"))
			}
			if src.sess != nil && !sess.resume() {
				return nil
			}
		}
	}
}
```

```go
// backFromAI 返回主模式；有未存档会话时询问存档（Esc 与 /back 共用）
func (r *REPL) backFromAI() {
	ctx := r.newCtx(nil)

	if err := command.MaybeSaveAIChat(ctx); err != nil {
		fmt.Fprintln(r.out, ui.Err("错误: "+err.Error()))

}
```

## 已知边界（实施时不再扩scope）

- `confirm`/`inputLine` 暂留 huh，其 Esc 中止会冒 `huh.ErrUserAborted` 英文报错（含 `/back` 存档询问场景）——并行会话已判定单字段无渲染 bug，本计划沿用其结论，如需处理另行立项。

- raw 模式下方向键等不可分发按键被静默忽略（`pagerKeySource.next` 内循环），不退出视图。
