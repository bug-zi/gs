```go
// maxCliLines 短文件阈值：不超过此行数进 CLI 内编辑，否则外部编辑器
const maxCliLines = 20

// shouldCliEdit 行数 ≤ maxCliLines 视为短文件
func shouldCliEdit(data []byte) bool { return countLines(data) <= maxCliLines }

// countLines 行数：\n 计数，末行无换行符补 1；空文件为 0
func countLines(data []byte) int {
	n := bytes.Count(data, []byte("\n"))
	if len(data) > 0 && data[len(data)-1] != '\n' {
		n++
	}
	return n
}
```

```go
package edit

import (
	"bytes"
	"strings"

	"github.com/bugzi/gs/internal/workspace"
	"github.com/charmbracelet/bubbles/textarea"
)

type inlineModel struct {
	ta      textarea.Model
	path    string
	crlf    bool   // 原文件是否 CRLF，保存时写回同风格
	initial string // 进入时内容，Esc 时判断是否改过
	err     error  // 保存失败信息，非 nil 时底部展示
	confirm bool   // Esc 后待确认「放弃修改」
}

func newInlineModel(path string, data []byte) inlineModel {
	crlf := bytes.Contains(data, []byte("\r\n"))
	ta := textarea.New()
	ta.SetValue(normalizeCRLF(data))
	ta.CharLimit = 0 // 不限长度
	ta.Prompt = ""
	ta.ShowLineNumbers = false
	ta.Focus()
	return inlineModel{ta: ta, path: path, crlf: crlf, initial: normalizeCRLF(data)}
}

func normalizeCRLF(data []byte) string { return strings.ReplaceAll(string(data), "\r\n", "\n") }

// saveFile 按 detect 出的换行风格原子写回（项目纪律：一律原子写）
func saveFile(path, content string, crlf bool) error {
	out := []byte(content)
	if crlf {
		out = bytes.ReplaceAll(out, []byte("\n"), []byte("\r\n"))
	}
	return workspace.WriteFileAtomic(path, out)
}
```

（若 `go mod tidy` 提示缺 bubbles 依赖，先跑 `go mod tidy`——bubbletea/bubbles 已在 go.mod，应无需新增。）

```go
// EditInline CLI 全屏编辑 path；tea 启动失败返回错误，由调用方降级外部编辑器
func EditInline(path string, data []byte) error {
	m := newInlineModel(path, data)
	if _, err := tea.NewProgram(m, tea.WithAltScreen()).Run(); err != nil {
		return err
	}
	return m.err
}

func (m inlineModel) Init() tea.Cmd { return nil }

func (m inlineModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.ta.SetWidth(msg.Width)
		m.ta.SetHeight(msg.Height - 3) // 标题行 + 空行 + 确认/错误行余量
		return m, nil
	case tea.KeyMsg:
		switch msg.String() {
		case "ctrl+s":
			if err := saveFile(m.path, m.ta.Value(), m.crlf); err != nil {
				m.err = err
				return m, nil
			}
			m.err = nil
			return m, tea.Quit
		case "esc":
			if m.confirm { // 确认态再按 Esc = 不退了
				m.confirm = false
				return m, nil
			}
			if m.ta.Value() != m.initial {
				m.confirm = true
				return m, nil
			}
			return m, tea.Quit
		}
		if m.confirm { // 确认态吞掉其余按键，不落进编辑区
			switch strings.ToLower(msg.String()) {
			case "y":
				return m, tea.Quit
			case "n":
				m.confirm = false
				return m, nil
			}
			return m, nil
		}
	}
	var cmd tea.Cmd
	m.ta, cmd = m.ta.Update(msg)
	return m, cmd
}

func (m inlineModel) View() string {
	var b strings.Builder
	fmt.Fprintf(&b, "编辑 %s · %s · %s\n\n", ui.Path(m.path), ui.Dim("Ctrl+S 保存退出"), ui.Dim("Esc 退出"))
	b.WriteString(m.ta.View())
	if m.confirm {
		b.WriteString("\n有未保存修改，放弃并退出？(y/N)")
	}
	if m.err != nil {
		fmt.Fprintf(&b, "\n保存失败: %v", m.err)
	}
	return b.String()
}
```
