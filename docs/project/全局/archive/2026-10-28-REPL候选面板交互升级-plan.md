**Goal:** 候选面板按使用频率排序、↑↓ 循环选择、唯一命中命令 1 秒后显示参数用法提示。

**Architecture:** 频率在启动时从历史文件统计并对命令表稳定重排（complete 包匹配逻辑零改动）；循环选择提取纯函数 `wrapSel`；延迟提示由 `complete.Result.Single/SingleHint` 驱动 rawReader 内的 `time.AfterFunc` 状态机，重绘与定时器回调共用互斥锁。

**Tech Stack:** Go 标准库（sort/time/sync）、现有 `internal/complete`、`internal/repl`、`internal/termkey` 自绘输入行。

**纪律：** 所有 git 操作由开发者完成，AI 不执行任何 git 命令；各任务末尾仅给提交建议。落地判定 = `go build ./...` + `go test ./...` 全绿。

```go
// wrapSel 面板选中项循环移动：↑ 在首行回绕到末行，↓ 在末行回绕到首行；
// 未选中（-1）时就近进入列表第一行；n<=0 无候选维持未选中
func wrapSel(sel, delta, n int) int {
	if n <= 0 {
		return -1
	}
	if delta < 0 {
		switch {
		case sel < 0:
			return 0
		case sel == 0:
			return n - 1
		default:
			return sel - 1
		}
	}
	if sel >= n-1 {
		return 0
	}
	return sel + 1
}
```

`internal/repl/rawreader.go`：

3.1 文件尾新增辅助函数：

```go
// delayedHintText 延迟提示行内容：命令名 + 参数提示；任一为空则不显示
func delayedHintText(single, singleHint string) string {
	if single == "" || singleHint == "" {
		return ""
	}
	return single + " " + singleHint
}
```

3.2 import 增 `"sync"`、`"time"`。

3.3 `Read` 整体改造——`res` 提为闭包外变量；draw 拆为持锁版 `drawLocked`；新增 `stopHint`/`settle`；四个退出路径全部停表。完整替换后的 `Read`：

```go
func (r *rawReader) Read() (string, error) {
	if r.fallback != nil {
		return r.fallback.Read()
	}
	raw, err := term.MakeRaw(int(os.Stdin.Fd()))
	if err != nil {
		fmt.Fprintln(os.Stderr, "（无法进入原始输入模式，已降级为普通输入）")
		r.fallback = newPlainReader(r.prompt)
		return r.fallback.Read()
	}
	defer func() {
		_ = term.Restore(int(os.Stdin.Fd()), raw)
		termkey.DrainConsoleInput()
	}()

	st := NewLineState()
	var res complete.Result
	var ghost, hint string
	var cands []string
	panel, sel, noPanel := false, -1, false
	histIdx, typed := -1, ""

	var mu sync.Mutex // 守护以下状态与重绘：定时器回调与主循环并发触达
	var hintTimer *time.Timer
	var hintText string // 非空 = 输入行下方延迟提示显示中

	stopHint := func() {
		if hintTimer != nil {
			hintTimer.Stop()
			hintTimer = nil
		}
		hintText = ""
	}

	// drawLocked 重绘整个帧（须持 mu）：清输入行及下方旧内容 → 输入行 → 面板/延迟提示 → 光标复位
	drawLocked := func() {
		res = complete.For(r.tree, st.Text())
		ghost, hint, cands = res.Ghost, res.Hint, res.Candidates
		panel = len(cands) >= 2 && !noPanel
		if !panel {
			sel = -1
		} else if sel >= len(cands) {
			sel = -1
		}
		var lines []string
		if panel {
			lines = buildPanelLines(cands, hint, sel)
		} else if hintText != "" {
			lines = []string{"  \x1b[2m" + hintText + "\x1b[0m"} // 延迟提示：暗色，与面板项同缩进
		}
		avail := terminalWidth() - runewidth.StringWidth(r.prompt) - 1
		line, curCol := buildInputLineFit(r.prompt, st, ghost, avail)
		var b strings.Builder
		b.WriteString("\r\x1b[J")
		b.WriteString(line)
		for _, l := range lines {
			b.WriteString("\r\n\x1b[K")
			b.WriteString(l)
		}
		if len(lines) > 0 {
			fmt.Fprintf(&b, "\x1b[%dA", len(lines))
		}
		fmt.Fprintf(&b, "\r\x1b[%dC", curCol)
		os.Stdout.WriteString(b.String())
	}

	// settle 每轮按键后的收尾（须持 mu）：清提示与定时器 → 重绘 → 唯一命中且面板未弹时设 1s 定时器
	settle := func() {
		stopHint()
		drawLocked()
		text := delayedHintText(res.Single, res.SingleHint)
		if text == "" || noPanel {
			return
		}
		line := st.Text()
		hintTimer = time.AfterFunc(time.Second, func() {
			mu.Lock()
			defer mu.Unlock()
			if st.Text() != line || noPanel { // 行已变或 Esc 压制面板：放弃
				return
			}
			hintText = text
			drawLocked()
		})
	}

	histMove := func(delta int) {
		items := r.hist.Items()
		if len(items) == 0 {
			return
		}
		if histIdx < 0 {
			if delta > 0 {
				return
			}
			typed = st.Text()
			histIdx = len(items) - 1
		} else {
			histIdx += delta
			if histIdx < 0 {
				histIdx = -1
				st = FromText(typed)
				st.Pos = st.Len()
				return
			}
			if histIdx > len(items)-1 {
				histIdx = len(items) - 1
			}
		}
		st = FromText(items[histIdx])
		st.Pos = st.Len()
	}

	var parser termkey.Parser
	mu.Lock()
	drawLocked()
	mu.Unlock()
	buf := make([]byte, 256)
	for {
		n, err := os.Stdin.Read(buf)
		if err != nil {
			mu.Lock()
			stopHint()
			mu.Unlock()
			return finishRead(r, st)
		}
		chunk := buf[:n]
		keys := parser.Feed(chunk)
		mu.Lock()
		for _, key := range keys {
			switch key.Kind {
			case termkey.KindRune:
				st.Insert(key.R)
			case termkey.KindBackspace:
				st.Backspace()
			case termkey.KindDelete:
				st.Delete()
			case termkey.KindLeft:
				st.Left()
			case termkey.KindRight:
				if st.Pos == st.Len() && ghost != "" {
					st.InsertText(ghost) // 行尾 → 接受 ghost
				} else {
					st.Right()
				}
			case termkey.KindHome:
				st.Pos = 0
			case termkey.KindEnd:
				st.Pos = st.Len()
			case termkey.KindTab:
				if ghost != "" && st.Pos == st.Len() {
					st.InsertText(ghost)
				}
			case termkey.KindEsc:
				if strings.HasPrefix(r.prompt, "ai") {
					os.Stdout.WriteString("\r\x1b[J\r\n") // 清输入帧（面板+ghost+提示）后交还普通终端
					stopHint()
					mu.Unlock()
					return "", ErrEscBack
				}
				noPanel = true
				panel = false
				sel = -1
			case termkey.KindCtrlC:
				st = NewLineState()
				histIdx, typed, noPanel, sel = -1, "", false, -1
			case termkey.KindCtrlD:
				if st.Len() == 0 {
					os.Stdout.WriteString("\r\x1b[J\r\n")
					stopHint()
					mu.Unlock()
					return "", io.EOF
				}
				st.Delete()
			case termkey.KindUp:
				if panel && len(cands) > 0 {
					sel = wrapSel(sel, -1, len(cands))
				} else {
					histMove(-1)
				}
			case termkey.KindDown:
				if panel && len(cands) > 0 {
					sel = wrapSel(sel, 1, len(cands))
				} else {
					histMove(1)
				}
			case termkey.KindEnter:
				if panel && sel >= 0 && sel < len(cands) {
					st = FromText(acceptCandidateText(st.Text(), cands[sel]))
					st.Pos = st.Len()
					noPanel, sel = false, -1
					drawLocked()
					continue
				}
				stopHint()
				r.hist.Add(st.Text())
				mu.Unlock()
				return finishRead(r, st)
			}
		}
		settle()
		mu.Unlock()
	}
}
```

要点（供实施者核对，不改变上述代码）：
- 定时器回调与主循环全部触达点（按键处理、drawLocked、histMove、st 重赋值）都在 `mu` 临界区内；主循环阻塞在 `os.Stdin.Read` 时不持锁，定时器可独立重绘。
- `st` 是闭包捕获变量，回调里 `st.Text()` 总是当前行文本，`st.Text() != line` 即「行已变」。
- Esc 压制（`noPanel=true`）同时压制延迟提示的触发与到点落笔。

```go
package repl

import (
	"sort"
	"strings"

	"github.com/bugzi/gs/internal/command"
)

// freqFromHistory 统计历史中各命令使用次数：/ 开头行取首词，
// 小写化后经 command.Find 归一别名到正式命令名；非命令行忽略
func freqFromHistory(items []string, cmds []command.Command) map[string]int {
	freq := make(map[string]int)
	for _, line := range items {
		if !strings.HasPrefix(line, "/") {
			continue
		}
		fields := strings.Fields(line[1:])
		if len(fields) == 0 {
			continue
		}
		if cmd := command.Find(cmds, strings.ToLower(fields[0])); cmd != nil {
			freq[cmd.Name]++
		}
	}
	return freq
}

// sortCmdsByFreq 按使用次数稳定降序重排命令表；并列与零次保持原序（默认序垫底）
func sortCmdsByFreq(cmds []command.Command, freq map[string]int) []command.Command {
	out := append([]command.Command(nil), cmds...)
	sort.SliceStable(out, func(i, j int) bool {
		return freq[out[i].Name] > freq[out[j].Name]
	})
	return out
}
```
