**实现期决策（本计划锁定）：**

1. **摘录范围合并为一步**：`/river add` 选中 sea 条目后直接创建预填「出处+全文」的 river 条目并自动打开编辑器——全文摘录=直接保存退出，选段摘录=编辑器内删减后保存，无需额外的范围确认步骤。
2. **导入中转**：`Study/sea/inbox/` 为投放目录，`/sea import` 逐个整理（起标题）后归位 `sea/clip/<标题>.md`；启动 banner 在 inbox 非空时提示一行。
3. **AI 存档命名**：`sea/ai/<YYYY-MM-DD-HHMM>.md`，标题取首轮提问前 30 字。
4. **泉水纪律**：`/spring add` 预填占位提示行「（用自己的话改写，写完删掉本行）」，只给编辑器手写入口，无粘贴通道。

```go
package command

import (
	"errors"
	"io/fs"
	"path/filepath"
	"strings"

	"github.com/bugzi/gs/internal/workspace"
)

// tierDirs 层名 → 目录
var tierDirs = map[string]func(*workspace.Workspace) string{
	"sea":    (*workspace.Workspace).SeaDir,
	"river":  (*workspace.Workspace).RiverDir,
	"spring": (*workspace.Workspace).SpringDir,
}

// ListTierFiles 返回指定层（sea/river/spring）下全部 .md 的工作区相对路径（正斜杠）
func ListTierFiles(ws *workspace.Workspace, tier string) ([]string, error) {
	dirFn, ok := tierDirs[tier]
	if !ok {
		return nil, errors.New("未知层: " + tier)
	}
	var out []string
	err := filepath.WalkDir(dirFn(ws), func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // 层目录缺失视为空
		}
		if d.IsDir() || !strings.EqualFold(filepath.Ext(d.Name()), ".md") {
			return nil
		}
		rel, err := filepath.Rel(ws.Root, path)
		if err != nil {
			return nil
		}
		out = append(out, filepath.ToSlash(rel))
		return nil
	})
	return out, err
}

// ListKnowledge 返回三层知识库全部 .md（sea → river → spring 顺序）
func ListKnowledge(ws *workspace.Workspace) ([]string, error) {
	var out []string
	for _, tier := range []string{"sea", "river", "spring"} {
		files, err := ListTierFiles(ws, tier)
		if err != nil {
			return nil, err
		}
		out = append(out, files...)
	}
	return out, nil
}
```

```go
package command

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/bugzi/gs/internal/workspace"
)

// validateTopic 主题即文件名：禁止路径分隔符与特殊字符
func validateTopic(topic string) error {
	if strings.TrimSpace(topic) == "" {
		return fmt.Errorf("主题不能为空")
	}
	if strings.ContainsAny(topic, `/\:*?"<>|`) || strings.Contains(topic, "..") {
		return fmt.Errorf("主题不能包含路径分隔符或特殊字符")
	}
	return nil
}

// CreateEntry 在指定层创建条目：# 主题 + 出处引用行 + 正文，原子写，返回工作区相对路径。
// source 为空则不写出处行。
func CreateEntry(ws *workspace.Workspace, tier, topic, source, body string) (string, error) {
	if err := validateTopic(topic); err != nil {
		return "", err
	}
	dirFn, ok := tierDirs[tier]
	if !ok {
		return "", fmt.Errorf("未知层: %s", tier)
	}
	path := filepath.Join(dirFn(ws), topic+".md")
	if _, err := os.Stat(path); err == nil {
		return "", fmt.Errorf("条目已存在: %s", ws.Rel(path))
	}
	var b strings.Builder
	fmt.Fprintf(&b, "# %s\n\n", topic)
	if source != "" {
		fmt.Fprintf(&b, "> 出处：%s\n\n", source)
	}
	b.WriteString(body)
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return "", err
	}
	if err := workspace.WriteFileAtomic(path, []byte(b.String())); err != nil {
		return "", err
	}
	return ws.Rel(path), nil
}
```

```go
// pickTierEntry 层内选条目：关键词过滤后单选；返回工作区相对路径
func pickTierEntry(ctx *Ctx, tier, kw, label string) (string, error) {
	files, err := ListTierFiles(ctx.WS, tier)
	if err != nil {
		return "", err
	}
	if kw != "" {
		kw = strings.ToLower(kw)
		var matched []string
		for _, f := range files {
			if strings.Contains(strings.ToLower(f), kw) {
				matched = append(matched, f)
			}
		}
		files = matched
	}
	switch len(files) {
	case 0:
		return "", fmt.Errorf("%s 层没有匹配的条目", tier)
	case 1:
		return files[0], nil
	}
	var pick string
	opts := make([]huhOption, 0, len(files))
	for _, f := range files {
		opts = append(opts, option(f, f))
	}
	if err := selectFrom(ctx, label, opts, &pick); err != nil {
		return "", err
	}
	return pick, nil
}

// runTierView 条目查阅态：pager 渲染，e 编辑该条目
func runTierView(ctx *Ctx, rel string) error {

			}
			return ui.Path(rel) + "\n" + ui.RenderMD(string(data))
		},
		func() (string, bool) { return abs, true })
}

// runSeaImport inbox → clip 整理：逐个预览、起标题、迁移
func runSeaImport(ctx *Ctx) error {
	inbox := filepath.Join(ctx.WS.SeaDir(), "inbox")
	entries, err := os.ReadDir(inbox)
	if err != nil {
		fmt.Fprintln(ctx.Out, ui.Dim("没有 inbox 目录（放文件到 Study/sea/inbox/ 后再执行）"))
		return nil
	}
	n := 0
	for _, e := range entries {
		if e.IsDir() || !strings.EqualFold(filepath.Ext(e.Name()), ".md") {
			continue
		}
		src := filepath.Join(inbox, e.Name())
		fmt.Fprintf(ctx.Out, "\n%s\n%s\n", ui.Path("Study/sea/inbox/"+e.Name()), previewOf(src, 10))
		def := strings.TrimSuffix(e.Name(), ".md")
		title, err := inputLine(ctx, "导入标题（回车默认 "+def+"）")
		if err != nil {
			return err
		}
		title = strings.TrimSpace(title)
		if title == "" {
			title = def
		}
		dst := filepath.Join(ctx.WS.SeaDir(), "clip", title+".md")
		if _, err := os.Stat(dst); err == nil {
			fmt.Fprintln(ctx.Out, ui.Dim("跳过: clip 下已存在 "+title+".md"))
			continue
		}
		if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
			return err
		}
		if err := os.Rename(src, dst); err != nil {
			return err
		}
		fmt.Fprintf(ctx.Out, "已导入 %s\n", ui.Path(ctx.WS.Rel(dst)))
		n++
	}
	if n == 0 {
		fmt.Fprintln(ctx.Out, ui.Dim("inbox 中没有待导入的 md"))
	} else {
		fmt.Fprintf(ctx.Out, "导入完成：%d 个\n", n)
	}
	return nil
}

// newTierCmd 生成 /sea /river /spring 的查看+edit 命令壳
func newTierCmd(name, short string, extraSubs []string, extra func(ctx *Ctx, sub string) (bool, error)) Command {
	subs := append([]string{"edit"}, extraSubs...)
	return Command{
		Name:  name,
		Short: short,
		Subs:  subs,
		Run: func(ctx *Ctx) error {
			if len(ctx.Args) > 0 {
				if handled, err := extra(ctx, ctx.Args[0]); handled || err != nil {
					return err
				}
				if ctx.Args[0] == "edit" {
					rel, err := pickTierEntry(ctx, name, strings.Join(ctx.Args[1:], " "), "选择要编辑的条目")
					if err != nil {
						return err
					}
					return editOpen(ctx, filepath.Join(ctx.WS.Root, filepath.FromSlash(rel)))
				}
			}
			rel, err := pickTierEntry(ctx, name, strings.Join(ctx.Args, " "), "选择条目")
			if err != nil {
				return err
			}
			return runTierView(ctx, rel)
		},
	}
}
```

```go
// askTopic 起主题名：输入框预填默认值
func askTopic(ctx *Ctx, def string) (string, error) {
	s, err := inputLine(ctx, "条目主题（回车默认 "+def+"）")
	if err != nil {
		return "", err
	}
	s = strings.TrimSpace(s)
	if s == "" {
		s = def
	}
	return s, nil
}

// runRiverAdd 大海→河流 摘录：选 sea 条目 → 创建预填「出处+全文」的 river 条目 → 打开编辑器（全文直存即全文摘录，删减后存即选段摘录）
func runRiverAdd(ctx *Ctx) error {
	src, err := pickTierEntry(ctx, "sea", strings.Join(ctx.Args, " "), "选择要摘录的大海条目")
	if err != nil {
		return err
	}
	data, err := os.ReadFile(filepath.Join(ctx.WS.Root, filepath.FromSlash(src)))
	if err != nil {
		return err
	}
	def := strings.TrimSuffix(filepath.Base(src), ".md")
	topic, err := askTopic(ctx, def)
	if err != nil {
		return err
	}
	rel, err := CreateEntry(ctx.WS, "river", topic, src, string(data))
	if err != nil {
		return err
	}
	fmt.Fprintf(ctx.Out, "已创建 %s（可在编辑器内删减为选段）\n", ui.Path(rel))
	return editOpen(ctx, filepath.Join(ctx.WS.Root, filepath.FromSlash(rel)))
}
```

```go
package command

import (
	"fmt"
	"strings"
	"time"

	"github.com/bugzi/gs/internal/ai"
	"github.com/bugzi/gs/internal/workspace"
)

// SaveAIChat 把当前 /AI 会话整理成 md 存入 sea/ai/，返回新文件的工作区相对路径。
// 标题取首轮提问前 30 字；文件名用存档时刻，避免主题非法字符问题。
func SaveAIChat(ws *workspace.Workspace, st *AIState) (string, error) {
	if st == nil || len(st.Hist) == 0 {
		return "", fmt.Errorf("当前会话没有可存档的内容")
	}
	title := "AI 问答"
	for _, m := range st.Hist {
		if m.Role == "user" {
			r := []rune(strings.TrimSpace(m.Content))
			if len(r) > 30 {
				r = r[:30]
			}
			if n := strings.TrimSpace(string(r)); n != "" {
				title = strings.ReplaceAll(n, "\n", " ")
			}
			break
		}
	}
	var b strings.Builder
	fmt.Fprintf(&b, "# %s\n\n", title)
	fmt.Fprintf(&b, "> 存档自 /AI 会话 · %s\n\n", time.Now().In(workspace.CST).Format("2006-01-02 15:04"))
	for _, m := range st.Hist {
		switch m.Role {
		case "user":
			fmt.Fprintf(&b, "## 问\n\n%s\n\n", strings.TrimSpace(m.Content))
		case "assistant":
			fmt.Fprintf(&b, "## 答\n\n%s\n\n", strings.TrimSpace(m.Content))
		}
	}
	name := time.Now().In(workspace.CST).Format("2006-01-02-1504")
	return CreateEntry(ws, "sea", name, "", b.String())
}
```

注意：CreateEntry 会再生成 `# <name>` 标题行——为避免双标题，body 不以 `#` 开头即可；这里 body 以 `# title` 开头会双标题。**修正**：name 作为 topic（`2006-01-02-1504`），body 改为不带 `# name`，直接从出处行后开始：把 `fmt.Fprintf(&b, "# %s\n\n", title)` 从 body 中去掉，标题行由 CreateEntry 生成（内容为 name）。为保留语义化标题，把标题放进引用行：`> 存档自 /AI 会话 · <时间> · <title>`。最终 body 组装：

```go
// InboxCount 统计 sea/inbox 下待导入的 md 数量
func InboxCount(ws *workspace.Workspace) int {
	entries, err := os.ReadDir(filepath.Join(ws.SeaDir(), "inbox"))
	if err != nil {
		return 0
	}
	n := 0
	for _, e := range entries {
		if !e.IsDir() && strings.EqualFold(filepath.Ext(e.Name()), ".md") {
			n++
		}
	}
	return n
}
```
