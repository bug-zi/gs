// 交互壳手绘 ANSI 共用色板：与 render.go 的 lipgloss 主题同族（220 金 → 179 赭 → 144 沙），低饱和、禁彩亮
const (
	ANSIGold      = "\x1b[38;5;220m\x1b[1m" // 金粗：品牌字/选中行箭头
	ANSIOchre     = "\x1b[38;5;179m"        // 赭：banner 标签
	ANSIOchreBold = "\x1b[38;5;179m\x1b[1m" // 赭粗：ai 提示符
	ANSIBright    = "\x1b[38;5;255m"        // 亮白：选中项文字
	ANSIMid       = "\x1b[38;5;245m"        // 中灰：未选项
	ANSIIdle      = "\x1b[38;5;240m"        // 暗灰：箭头/提示行
	ANSISelBg     = "\x1b[48;5;236m"        // 选中行微底
	ANSIItalic    = "\x1b[3m"               // 斜体
	ANSIReset     = "\x1b[0m"
)
```

// buildPanelLines 候选面板行：提示斜体暗灰在最上，选中项金箭头+微底亮字，未选中中灰
func buildPanelLines(cands []string, hint string, sel int) []string {
	var lines []string
	if hint != "" {
		lines = append(lines, ui.ANSIIdle+ui.ANSIItalic+hint+ui.ANSIReset)
	}
	for i, c := range cands {
		if i == sel {
			lines = append(lines, ui.ANSIGold+"❯"+ui.ANSIReset+" "+ui.ANSISelBg+ui.ANSIBright+c+ui.ANSIReset)
		} else {
			lines = append(lines, "  "+ui.ANSIMid+c+ui.ANSIReset)
		}
	}
	return lines
}
```

// plainWidth 剥离 ANSI CSI 转义序列后的显示宽度（带色提示符计宽用）
func plainWidth(s string) int {
	var b strings.Builder
	for i := 0; i < len(s); {
		if s[i] == 0x1b && i+1 < len(s) && s[i+1] == '[' {
			j := i + 2
			for j < len(s) && !((s[j] >= 'a' && s[j] <= 'z') || (s[j] >= 'A' && s[j] <= 'Z')) {
				j++
			}
			if j < len(s) {
				j++
			}
			i = j
			continue
		}
		b.WriteByte(s[i])
		i++
	}
	return runewidth.StringWidth(b.String())
}
```

func (r *rawReader) SetPrompt(p string) {
	r.prompt, r.promptName = brandPrompt(p)
}

// brandPrompt 提示符品牌色：gs 金粗、ai 赭粗，其余（箭头段）暗灰；返回带色串与素文本首词
func brandPrompt(p string) (string, string) {
	i := strings.Index(p, " ")
	if i <= 0 {
		return p, p
	}
	name, rest := p[:i], p[i:]
	color := ui.ANSIGold
	if name == "ai" {
		color = ui.ANSIOchreBold
	}
	return color + name + ui.ANSIReset + ui.ANSIIdle + rest + ui.ANSIReset, name
}
```
