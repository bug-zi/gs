```go
package tier

import (
	"fmt"
	"os"
	"path/filepath"
	"time"

	"github.com/bugzi/gs/internal/workspace"
)

// TrashDirRel 回收站根（工作区相对）。位于 .gs/ 下，不在 Study/、Project/ 内容区，
// 搜索经默认 scan.exclude 的 ".gs" 排除。
const TrashDirRel = ".gs/trash"

const trashReadme = `# gs 回收站

此处存放 gs 删除机制移入的条目（软删除，不自动清理）。

目录结构：<删除时间戳>-<序号>/<原工作区相对路径>。

找回方法：把对应文件/目录 mv 回原相对路径即可，例如

    mv .gs/trash/20261028-180000-1/Study/river/a.md Study/river/a.md

条目删除时绑定已清；找回后可重新挂载（/plan 浏览器 a 键）。
`

// MoveToTrash 软删除：把 rel（文件或目录）移入 <root>/.gs/trash/<YYYYMMDD-HHMMSS>-<n>/<rel>，
// 保留原相对结构。同秒多次删除以 n 递增防碰撞；首次使用时写 README 找回说明。
// 返回回收站内工作区相对路径。
func MoveToTrash(root, rel string, now time.Time) (string, error) {
	src := filepath.Join(root, filepath.FromSlash(rel))
	if _, err := os.Stat(src); err != nil {
		return "", fmt.Errorf("源不存在: %s", rel)
	}
	stamp := now.In(workspace.CST).Format("20060102-150405")
	for n := 1; ; n++ {
		dstRel := fmt.Sprintf("%s/%s-%d/%s", TrashDirRel, stamp, n, rel)
		dst := filepath.Join(root, filepath.FromSlash(dstRel))
		if _, err := os.Lstat(dst); err == nil {
			continue // 同秒同路径已占用，顺延序号
		}
		if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
			return "", err
		}
		if err := os.Rename(src, dst); err != nil {
			return "", err
		}
		writeTrashReadme(root)
		return dstRel, nil
	}
}

// writeTrashReadme 首次使用时写找回说明；已存在则跳过（失败不影响删除本身）
func writeTrashReadme(root string) {
	p := filepath.Join(root, filepath.FromSlash(TrashDirRel), "README.md")
	if _, err := os.Stat(p); err == nil {
		return
	}
	_ = workspace.WriteFileAtomic(p, []byte(trashReadme))
}
```

```go
// OwnersOf 反查绑定路径所属的全部知识点（树序；spring/code 单值与 sea/ship/river 清单均查）
func (rm *Roadmap) OwnersOf(rel string) []PointRef {
	var out []PointRef
	for bi := range rm.Branches {
		for pi := range rm.Branches[bi].Points {
			p := &rm.Branches[bi].Points[pi]
			if p.Spring == rel || p.Code == rel {
				out = append(out, PointRef{bi, pi})
				continue
			}
			hit := false
			for _, list := range [][]string{p.Sea, p.Ship, p.River} {
				for _, s := range list {
					if s == rel {
						hit = true
						break
					}
				}
				if hit {
					break
				}
			}
			if hit {
				out = append(out, PointRef{bi, pi})
			}
		}
	}
	return out
}

// DetachPoint 清空知识点五类绑定，返回清出的 river 绑定清单（供壳层收容迁移）。
// 只动内存不落盘；清空后 PointEmpty 为真，DeletePoint 可直接调用。
func (rm *Roadmap) DetachPoint(r PointRef) []string {
	p := rm.PointAt(r)
	river := append([]string{}, p.River...)
	p.Spring, p.Code = "", ""
	p.Sea, p.Ship, p.River = nil, nil, nil
	return river
}

// DeleteBranch 删除空主干；仍有知识点或仅剩最后一个主干时拒绝（壳层先提示处理）
func (rm *Roadmap) DeleteBranch(bi int) error {
	b := &rm.Branches[bi]
	if len(b.Points) > 0 {
		return fmt.Errorf("主干「%s」仍有 %d 个知识点，先删除或移走后再删除主干", b.Title, len(b.Points))
	}
	if len(rm.Branches) <= 1 {
		return errors.New("至少保留一个主干")
	}
	rm.Branches = append(rm.Branches[:bi], rm.Branches[bi+1:]...)
	return nil
}
```

```go
// BuildUnmountMoves 把 rels 收容到 99-未挂载（保留原名；与收容区现有文件撞名时基名后插 .k 顺延）。
// existingUnmounted 为收容区现有文件名清单，纯函数不触碰文件系统。
func BuildUnmountMoves(rels []string, existingUnmounted []string) []FileMove {
	used := map[string]bool{}
	for _, e := range existingUnmounted {
		used[e] = true
	}
	moves := make([]FileMove, 0, len(rels))
	for _, rel := range rels {
		base := filepath.Base(rel)
		if used[base] {
			ext := filepath.Ext(base)
			stem := strings.TrimSuffix(base, ext)
			for k := 1; ; k++ {
				cand := fmt.Sprintf("%s.%d%s", stem, k, ext)
				if !used[cand] {
					base = cand
					break
				}
			}
		}
		used[base] = true
		moves = append(moves, FileMove{rel, RiverDirRel + "/" + UnmountedDirName + "/" + base})
	}
	return moves
}

// ListUnmounted 列出收容区现有文件名（目录缺失视为空；子目录不计入）
func ListUnmounted(root string) []string {
	entries, err := os.ReadDir(filepath.Join(root, RiverDirRel, UnmountedDirName))
	if err != nil {
		return nil
	}
	var out []string
	for _, e := range entries {
		if !e.IsDir() {
			out = append(out, e.Name())
		}
	}
	return out
}
```

```go
// CountPaths 统计指向 paths 集合中路径的 files 条目数
func (ix *Index) CountPaths(paths map[string]bool) int {
	n := 0
	for ti := range ix.Topics {
		for fi := range ix.Topics[ti].Files {
			if paths[ix.Topics[ti].Files[fi].Path] {
				n++
			}
		}
	}
	return n
}

// PrunePaths 删除指向 paths 集合中路径的 files 条目；返回清除条数（不落盘）
func (ix *Index) PrunePaths(paths map[string]bool) int {
	n := 0
	for ti := range ix.Topics {
		kept := ix.Topics[ti].Files[:0]
		for _, f := range ix.Topics[ti].Files {
			if paths[f.Path] {
				n++
				continue
			}
			kept = append(kept, f)
		}
		ix.Topics[ti].Files = kept
```

```go
// runTierDel /sea|ship|river|spring|code del 入口：选择条目 → 删除全流程
func runTierDel(ctx *Ctx, tierName, kw string) error {
	it, err := pickTierEntryGrouped(ctx, tierName, kw, "选择要删除的条目")
	if err != nil {
		return err
	}
	_, _ = deleteTierEntry(ctx, tierName, it.Path)
	return nil
}

// deleteTierEntry 条目删除全流程：预览影响 → y/N 确认 → 落盘绑定 → 回收站 → 引用清理。
// 返回是否已删除；输入失败（Esc 等）与非 y 一律视为取消不报错——查阅态 [x] 键复用本流程，
// 取消应静默回到查阅。
func deleteTierEntry(ctx *Ctx, tierName, rel string) (bool, error) {
	printTierDeletePreview(ctx, tierName, rel)
	ans, err := inputLine(ctx, "确认删除？（y/N）")
	if err != nil || strings.TrimSpace(ans) != "y" && strings.TrimSpace(ans) != "Y" {
		fmt.Fprintln(ctx.Out, "已取消")
		return false, nil
	}
	if err := applyTierDelete(ctx, tierName, rel); err != nil {
		return false, err
	}
	return true, nil
}

// printTierDeletePreview 输出删除影响预览：文件迁移、绑定清理、引用统计
func printTierDeletePreview(ctx *Ctx, tierName, rel string) {
	fmt.Fprintf(ctx.Out, "删除 %s 条目：%s\n", tierName, ui.Path(rel))
	fmt.Fprintln(ctx.Out, ui.Dim("  文件将移入回收站（.gs/trash/<时间戳>/ 下保留原相对结构）"))
	owners := []roadmap.PointRef{}
	if rm, err := loadRoadmapSoft(ctx); err == nil && rm != nil {
		owners = rm.OwnersOf(rel)
		if len(owners) > 0 {
			fmt.Fprintln(ctx.Out, "绑定清理：")
			for _, o := range owners {
				fmt.Fprintf(ctx.Out, "  - %s（%s）\n", rm.PathOf(o), tierName)
			}
		}
	}
	refs := false
	if n := countTopicsRefs(ctx, rel); n > 0 {
		refs = true
		fmt.Fprintf(ctx.Out, "引用统计：topics.toml 将清除 %d 条\n", n)
	}
	if n := tier.CountChainRefs(ctx.WS.Root, rel); n > 0 {
		refs = true
		fmt.Fprintf(ctx.Out, "引用统计：出处链引用 %d 处将失效（原文保留不动）\n", n)
	}
	if len(owners) == 0 && !refs {
		fmt.Fprintln(ctx.Out, ui.Dim("无绑定与引用，仅移入回收站"))
	}
}

// applyTierDelete 删除执行：清绑定落盘 → 回收站迁移 → topics 清理。
// 出处链指向被删文件的一律只警告不改写（不擅自改内容）；迁移失败时绑定已清、
// 文件原地保留，重跑 del 幂等修复。
func applyTierDelete(ctx *Ctx, tierName, rel string) error {
	rm, err := loadRoadmapSoft(ctx)
	if err != nil {
		return err
	}
	if rm != nil {
		if owners := unbindEntry(rm, tierName, rel); len(owners) > 0 {
			if err := roadmap.Save(ctx.WS.RoadmapFile(), rm); err != nil {
				return fmt.Errorf("绑定落盘失败（文件未迁移）: %w", err)
			}
			for _, o := range owners {
				fmt.Fprintf(ctx.Out, "%s 已解绑 %s\n", ui.OK("绑定"), rm.PathOf(o))
			}
		}
	}
	trashRel, err := tier.MoveToTrash(ctx.WS.Root, rel, timeNowCST())
	if err != nil {
		return fmt.Errorf("回收站迁移失败（绑定已清，文件原地保留，可重跑 del 幂等修复）: %w", err)
	}
	fmt.Fprintf(ctx.Out, "%s 已移入回收站 %s\n", ui.OK("删除"), ui.Dim(trashRel))
	pruneTopics(ctx, rel)
	if n := tier.CountChainRefs(ctx.WS.Root, rel); n > 0 {
		fmt.Fprintf(ctx.Out, ui.Dim("注意：出处链引用 %d 处已失效（原文保留）")+"\n", n)
	}
	return nil
}

// unbindEntry 从全部挂载点清除该条目的绑定；返回受影响知识点
func unbindEntry(rm *roadmap.Roadmap, tierName, rel string) []roadmap.PointRef {
	owners := rm.OwnersOf(rel)
	for _, o := range owners {
		switch tierName {
		case "spring":
			rm.UnbindSpring(o)
		case "code":
			rm.UnbindCode(o)
		case "sea":
			rm.UnbindList(o, roadmap.SeaList, rel)
		case "ship":
			rm.UnbindList(o, roadmap.ShipList, rel)
		case "river":
			rm.UnbindList(o, roadmap.RiverList, rel)
		}
	}
	return owners
}

// countTopicsRefs topics.toml 中指向 rel 的条目数；索引不存在或读取失败视为 0
func countTopicsRefs(ctx *Ctx, rel string) int {
	if _, err := os.Stat(ctx.WS.IndexPath()); err != nil {
		return 0
	}
	ix, err := index.Load(ctx.WS.IndexPath())
	if err != nil {
		return 0
	}
	return ix.CountPaths(map[string]bool{rel: true})
}

// pruneTopics 清除 topics.toml 中指向 rel 的条目并落盘；索引不存在时 no-op
func pruneTopics(ctx *Ctx, rel string) {
	if _, err := os.Stat(ctx.WS.IndexPath()); err != nil {
		return
	}
	ix, err := index.Load(ctx.WS.IndexPath())
	if err != nil {
		fmt.Fprintln(ctx.Out, ui.Err("topics.toml 读取失败: "+err.Error()))
		return
	}
	if n := ix.PrunePaths(map[string]bool{rel: true}); n > 0 {
		if err := ix.Save(ctx.WS.IndexPath()); err != nil {
			fmt.Fprintln(ctx.Out, ui.Err("topics.toml 落盘失败: "+err.Error()))
			return
		}
		fmt.Fprintf(ctx.Out, "topics.toml 清除 %d 条\n", n)
	}
}

// applyFileMoves 执行文件迁移并同步改写 topics 与出处链；返回出处链改写处数。
// planview 删点收容与 /plan del point 共用；迁移语义同 renumber（ApplyRenumber 幂等可重跑）。
func applyFileMoves(ctx *Ctx, moves []roadmap.FileMove) (int, error) {
	if len(moves) == 0 {
		return 0, nil
	}
	if err := tier.ApplyRenumber(ctx.WS.Root, &roadmap.RenumberPlan{Moves: moves}); err != nil {
		return 0, err
	}
	if n, err := rewriteTopics(ctx.WS, moves); err != nil {
		fmt.Fprintln(ctx.Out, ui.Err("topics.toml 改写失败: "+err.Error()))
	} else if n > 0 {
		fmt.Fprintf(ctx.Out, "topics.toml 改写 %d 处\n", n)
	}
	chainN := 0
	for _, m := range moves {
		if n, err := tier.RewriteSourceChains(ctx.WS.Root, m.From, m.To); err == nil {
			chainN += n
		}
	}
	return chainN, nil
}
```

```go
// runPlanDel /plan del branch|point <关键词>：非交互直达删除
func runPlanDel(ctx *Ctx, args []string) error {

func applyPointDelete(ctx *Ctx, rm *roadmap.Roadmap, ref roadmap.PointRef, river []string) error {

	title := b.Title
	if err := rm.DeleteBranch(bi); err != nil {
		return err
	}
	if err := roadmap.Save(ctx.WS.RoadmapFile(), rm); err != nil {
		return fmt.Errorf("roadmap 落盘失败: %w", err)
	}
	fmt.Fprintf(ctx.Out, "%s 主干「%s」已删除\n", ui.OK("完成"), title)
	dirRel := fmt.Sprintf("%s/%02d-%s", roadmap.RiverDirRel, bi+1, title)
	if entries, err := os.ReadDir(filepath.Join(ctx.WS.Root, filepath.FromSlash(dirRel))); err == nil && len(entries) == 0 {
		if trashRel, err := tier.MoveToTrash(ctx.WS.Root, dirRel, timeNowCST()); err == nil {
			fmt.Fprintf(ctx.Out, "空板块目录已入回收站 %s\n", ui.Dim(trashRel))
		}
	}
	fmt.Fprintln(ctx.Out, ui.Dim("可运行 /plan renumber 重排其余板块编号"))
	return nil
}

// locateBranch 主干定位：空关键词全量弹选；关键词唯一直达、多匹配弹选
func locateBranch(ctx *Ctx, rm *roadmap.Roadmap, kw string) (int, error) {
	if kw == "" {
		return pickBranchIdx(ctx, rm)
	}
	ms := rm.MatchBranches(kw)
	switch len(ms) {
	case 0:
		return 0, fmt.Errorf("没有匹配 %q 的主干", kw)
	case 1:
		return ms[0], nil
	}
	return pickBranchFrom(ctx, rm, ms, "匹配到多个主干")
}
```
