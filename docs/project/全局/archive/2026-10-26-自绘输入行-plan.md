```go
package complete

import "strings"

// Result 输入行当前应展示的补全状态
type Result struct {
	Ghost      string   // 可接受续写（Tab/→ 上屏）；空 = 无
	Candidates []string // 面板候选；空 = 不弹
	Hint       string   // 面板顶部参数提示；空 = 无
}

// For 计算输入行的补全状态。匹配大小写不敏感；
// 别名与其命令名同时匹配前缀时只保留命令名。
func For(tree Tree, line string) Result {
	if !strings.HasPrefix(line, "/") {
		return Result{}
	}
	body := line[1:]
	if body == "" {
		return Result{Candidates: allNames(tree)}
	}
	fields := strings.Split(body, " ")
	first := fields[0]
	if len(fields) == 1 {
		cands := matchNames(tree, first)
		if len(cands) == 1 {
			return Result{Ghost: ghostOf(cands[0], first)}
		}
		if len(cands) > 1 {
			return Result{Candidates: cands}
		}
		return Result{}
	}
	cmd := findCmd(tree, first)
	if cmd == nil {
		return Result{}
	}
	if len(fields) == 2 && fields[1] != "" {
		second := fields[1]
		var cands []string
		for _, s := range cmd.Subs {
			if strings.HasPrefix(strings.ToLower(s), strings.ToLower(second)) {
				cands = append(cands, s)
			}
		}
		if len(cands) == 1 {
			return Result{Ghost: cands[0][len(second):], Hint: cmd.Hint}
		}
		if len(cands) > 1 {
			return Result{Candidates: cands, Hint: cmd.Hint}
		}
		return Result{Hint: cmd.Hint}
	}
	// 尾随空格或更深的参数位：列子命令 + 参数提示
	return Result{Candidates: append([]string(nil), cmd.Subs...), Hint: cmd.Hint}
}

// ghostOf 返回 full 相对已打部分 typed 的剩余；大小写不敏感校验前缀
func ghostOf(full, typed string) string {
	if !strings.HasPrefix(strings.ToLower(full), strings.ToLower(typed)) {
		return ""
	}
	return full[len(typed):]
}

// allNames 全部命令名（不含别名，避免面板冗余）
func allNames(tree Tree) []string {
	var out []string
	for _, c := range tree {
		out = append(out, c.Name)
	}
	return out
}

// matchNames 前缀匹配命令名；别名仅在其命令名不匹配时参与
func matchNames(tree Tree, prefix string) []string {
	low := strings.ToLower(prefix)
	var out []string
	seen := map[string]bool{}
	add := func(n string) {
		if !seen[n] && strings.HasPrefix(strings.ToLower(n), low) {
			seen[n] = true
			out = append(out, n)
		}
	}
	for _, c := range tree {
		add(c.Name)
	}
	for _, c := range tree {
		nameMatch := strings.HasPrefix(strings.ToLower(c.Name), low)
		if nameMatch {
			continue // 命令名已匹配，别名不重复出现
		}
		for _, a := range c.Alias {
			add(a)
		}
	}
	return out
}

// findCmd 按名称或别名查找（大小写不敏感）
func findCmd(tree Tree, name string) *Cmd {
	low := strings.ToLower(name)
	for i := range tree {
		if strings.ToLower(tree[i].Name) == low {
			return &tree[i]
		}
	}
	for i := range tree {
		for _, a := range tree[i].Alias {
			if strings.ToLower(a) == low {
				return &tree[i]
			}
		}
	}
	return nil
}
```

```go
package repl

import (
	"bufio"
	"fmt"
	"os"
	"strings"
	"sync"
)

// History REPL 行历史：内存 + 追加持久化（每行一条，与原 readline 历史文件同格式）
type History struct {
	mu    sync.Mutex
	items []string
	file  string
}

// LoadHistory 读取历史文件；缺失/损坏从空开始
func LoadHistory(path string) *History {
	h := &History{file: path}
	f, err := os.Open(path)
	if err != nil {
		return h
	}
	defer f.Close()
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		if line := strings.TrimRight(sc.Text(), "\r"); line != "" {
			h.items = append(h.items, line)
		}
	}
	return h
}

// Items 返回历史快照（旧→新）
func (h *History) Items() []string {
	h.mu.Lock()
	defer h.mu.Unlock()
	return append([]string(nil), h.items...)
}

// Add 追加一条并持久化；空白与相邻重复忽略
func (h *History) Add(s string) {
	h.mu.Lock()
	defer h.mu.Unlock()
	if strings.TrimSpace(s) == "" {
		return
	}
	if n := len(h.items); n > 0 && h.items[n-1] == s {
		return
	}
	h.items = append(h.items, s)
	if h.file == "" {
		return
	}
	f, err := os.OpenFile(h.file, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o600)
	if err == nil {
		fmt.Fprintf(f, "%s\n", s)
		f.Close()
	}
}
```

```go
package repl

import (
	"io"
	"os"
	"strings"

	"github.com/bugzi/gs/internal/complete"
	"github.com/bugzi/gs/internal/ui"
	"github.com/charmbracelet/bubbles/textinput"
	tea "github.com/charmbracelet/bubbletea"
	"github.com/charmbracelet/lipgloss"
)

var (
	ghostStyle = lipgloss.NewStyle().Faint(true)
	selStyle   = lipgloss.NewStyle().Reverse(true)
)

// teaReader 自绘输入行：每次 Read 起一个内联 tea 程序，实现 Reader 接口
type teaReader struct {
	prompt   string
	tree     complete.Tree
	hist     *History
	fallback Reader // tea 启动失败时降级，之后一直走降级
}

func newTeaReader(prompt string, tree complete.Tree, hist *History) *teaReader {
	return &teaReader{prompt: prompt, tree: tree, hist: hist}
}

func (r *teaReader) SetPrompt(p string) { r.prompt = p }
func (r *teaReader) Close() error       { return nil }

func (r *teaReader) Read() (string, error) {
	if r.fallback != nil {
		return r.fallback.Read()
	}
	m := newLineModel(r.prompt, r.tree, r.hist)
	p := tea.NewProgram(m, tea.WithOutput(os.Stdout))
	if _, err := p.Run(); err != nil {
		fmt.Fprintln(os.Stderr, ui.Dim("（自绘输入行不可用，已降级为普通输入）"))
		r.fallback = newPlainReader(r.prompt)
		return r.fallback.Read()
	}
	if m.eof {
		return "", io.EOF
	}
	return m.result, nil
}

// lineModel 单行输入的 tea model
type lineModel struct {
	prompt string
	tree   complete.Tree
	hist   *History

	input   textinput.Model
	ghost   string
	cands   []string
	hint    string
	panel   bool // 面板是否显示
	sel     int  // 面板高亮，-1 无
	noPanel bool // Esc 后本行内不再弹面板

	histIdx int    // -1 = 不在历史浏览
	typed   string // 进入历史浏览前用户已打的内容

	result string
	eof    bool
}

func newLineModel(prompt string, tree complete.Tree, hist *History) lineModel {
	in := textinput.New()
	in.Prompt = "" // 提示符由 View 自行渲染
	in.Focus()
	return lineModel{
		prompt:  prompt,
		tree:    tree,
		hist:    hist,
		input:   in,
		sel:     -1,
		histIdx: -1,
	}
}

func (m lineModel) Init() tea.Cmd { return textinput.Blink }

func (m lineModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	key, ok := msg.(tea.KeyMsg)
	if !ok {
		return m, nil
	}
	switch key.Type {
	case tea.KeyCtrlC:
		m.input.SetValue("")
		m.histIdx = -1
		m.refresh()
		return m, nil
	case tea.KeyCtrlD:
		if m.input.Value() == "" {
			m.eof = true
			return m, tea.Quit
		}
		return m, nil
	case tea.KeyEnter:
		if m.panel && m.sel >= 0 && m.sel < len(m.cands) {
			m.acceptCandidate(m.cands[m.sel])
			m.refresh()
			return m, nil
		}
		m.result = m.input.Value()
		m.hist.Add(m.result)
		return m, tea.Quit
	case tea.KeyUp:
		if m.panel && len(m.cands) > 0 {
			if m.sel < 0 {
				m.sel = 0
			} else if m.sel > 0 {
				m.sel--
			}
			return m, nil
		}
		m.histPrev()
		m.refresh()
		return m, nil
	case tea.KeyDown:
		if m.panel && len(m.cands) > 0 {
			if m.sel < len(m.cands)-1 {
				m.sel++
			}
			return m, nil
		}
		m.histNext()
		m.refresh()
		return m, nil
	case tea.KeyEsc:
		if m.panel {
			m.noPanel = true
			m.panel = false
			m.sel = -1
			return m, nil
		}
	case tea.KeyTab:
		if m.acceptGhost() {
			m.refresh()
		}
		return m, nil
	case tea.KeyRight:
		if m.atEnd() && m.ghost != "" {
			m.acceptGhost()
			m.refresh()
			return m, nil
		}
	}
	// 其余按键交给 textinput（含中文输入）
	var cmd tea.Cmd
	m.input, cmd = m.input.Update(msg)
	m.histIdx = -1 // 重新打字即退出历史浏览
	m.refresh()
	return m, cmd
}

func (m lineModel) View() string {
	var b strings.Builder
	b.WriteString(m.prompt)
	b.WriteString(m.input.View())
	if m.ghost != "" && m.atEnd() {
		b.WriteString(ghostStyle.Render(m.ghost))
	}
	if m.panel {
		b.WriteString("\n")
		if m.hint != "" {
			b.WriteString(ui.Dim(m.hint))
			b.WriteString("\n")
		}
		for i, c := range m.cands {
			if i == m.sel {
				b.WriteString(selStyle.Render("❯ "+c))
			} else {
				b.WriteString("  " + ui.Dim(c))
			}
			if i < len(m.cands)-1 {
				b.WriteString("\n")
			}
		}
	}
	return b.String()
}

// refresh 根据当前输入重算补全状态
func (m *lineModel) refresh() {
	res := complete.For(m.tree, m.input.Value())
	m.ghost = res.Ghost
	m.hint = res.Hint
	m.cands = res.Candidates
	m.panel = len(m.cands) >= 2 && !m.noPanel
	if !m.panel {
		m.sel = -1
	} else if m.sel >= len(m.cands) {
		m.sel = -1
	}
}

func (m *lineModel) acceptGhost() bool {
	if m.ghost == "" {
		return false
	}
	m.input.SetValue(m.input.Value() + m.ghost)
	m.input.CursorEnd()
	return true
}

// acceptCandidate 用候选替换当前词：第一词阶段补 "/"+词，第二词阶段保留第一词
func (m *lineModel) acceptCandidate(c string) {
	v := m.input.Value()
	if i := strings.Index(v, " "); i >= 0 {
		m.input.SetValue(v[:i+1] + c)
	} else {
		m.input.SetValue("/" + c)
	}
	m.input.CursorEnd()
}

func (m *lineModel) histPrev() {
	items := m.hist.Items()
	if len(items) == 0 {
		return
	}
	if m.histIdx < 0 {
		m.typed = m.input.Value()
		m.histIdx = len(items) - 1
	} else if m.histIdx > 0 {
		m.histIdx--
	} else {
		return
	}
	m.input.SetValue(items[m.histIdx])
	m.input.CursorEnd()
}

func (m *lineModel) histNext() {
	if m.histIdx < 0 {
		return
	}
	items := m.hist.Items()
	if m.histIdx < len(items)-1 {
		m.histIdx++
		m.input.SetValue(items[m.histIdx])
	} else {
		m.histIdx = -1
		m.input.SetValue(m.typed)
	}
	m.input.CursorEnd()
}

func (m *lineModel) atEnd() bool {
	return m.input.Position() == len([]rune(m.input.Value()))
}
```

```go
package repl

import (
	"bufio"
	"os"

	"github.com/bugzi/gs/internal/complete"
	"golang.org/x/term"
)

// Reader 是 REPL 的行输入抽象
type Reader interface {
	Read() (string, error) // Ctrl-D 返回 io.EOF
	Close() error
	SetPrompt(p string)
}

// NewReader 默认自绘输入行；GS_PLAIN_INPUT=1 或 stdout 非 TTY 时降级 plain
func NewReader(prompt string, tree complete.Tree, hist *History) Reader {

		return newPlainReader(prompt)
	}

type plainReader struct {

	prompt string
}

}

func (r *plainReader) Read() (string, error) {
	os.Stdout.WriteString(r.prompt)
	line, err := r.br.ReadString('\n')
	if err != nil {
		return "", err
	}
	return strings.TrimRight(line, "\r\n"), nil
}
```
