```go
const undoLimit = 100

// snapshot 一次可回退状态:整份行拷贝 + 光标 + 粘滞列
type snapshot struct {
	lines    [][]rune
	cur, col int
	want     int
}

func cloneLines(lines [][]rune) [][]rune {
	out := make([][]rune, len(lines))
	for i, l := range lines {
		out[i] = append([]rune{}, l...)
	}
	return out
}

func (b *Buffer) snap() snapshot {
	return snapshot{lines: cloneLines(b.lines), cur: b.cur, col: b.col, want: b.want}
}

func (b *Buffer) restore(s snapshot) {
	b.lines = cloneLines(s.lines)
	b.cur, b.col, b.want = s.cur, s.col, s.want
}

// PushUndo 压入当前状态快照;命令层在每个可撤销动作前调用一次,同时清空 redo
func (b *Buffer) PushUndo() {
	b.undo = append(b.undo, b.snap())
	if len(b.undo) > undoLimit {
		b.undo = b.undo[1:]
	}
	b.redo = nil
}

// Undo 回退一步;空栈返回 false
func (b *Buffer) Undo() bool {
	if len(b.undo) == 0 {
		return false
	}
	s := b.undo[len(b.undo)-1]
	b.undo = b.undo[:len(b.undo)-1]
	b.redo = append(b.redo, b.snap())
	b.restore(s)
	return true
}

// Redo 重做一步;空栈返回 false
func (b *Buffer) Redo() bool {
	if len(b.redo) == 0 {
		return false
	}
	s := b.redo[len(b.redo)-1]
	b.redo = b.redo[:len(b.redo)-1]
	b.undo = append(b.undo, b.snap())
	b.restore(s)
	return true
}
```

要点：snap/restore 双向深拷贝——`Insert` 走 `append` 会原地改写行切片底层数组，浅拷贝快照会被别名污染（TestUndoSnapshotIsolation 卡的就是这个）。

```go
package edit

const killLimit = 20

// killRing 删除内容环形留存;Ctrl+Y 粘贴最近条目,连按向更早轮换
type killRing struct {
	items []string // 旧→新
}

func (k *killRing) push(s string) {
	if s == "" {
		return
	}
	if len(k.items) >= killLimit {
		k.items = k.items[1:]
	}
	k.items = append(k.items, s)
}

// yank 返回最近条目往回数 n 的条目(0=最近),模长回绕;空环 false
func (k *killRing) yank(n int) (string, bool) {
	if len(k.items) == 0 {
		return "", false
	}
	m := len(k.items)
	return k.items[((m-1-n)%m+m)%m], true
}
```

```go
package edit

import "strings"

// selPos 选区端点:逻辑行下标 + 行内 rune 下标
type selPos struct {
	line, col int
}

// normSel 归一化选区:起点≤终点(行序优先,行内列序)
func normSel(a, b selPos) (start, end selPos) {
	if a.line < b.line || (a.line == b.line && a.col <= b.col) {
		return a, b
	}
	return b, a
}

// selText 选区文本:起终行间以 \n 拼接(不含结尾换行)
func selText(lines [][]rune, start, end selPos) string {
	if start.line == end.line {
		return string(lines[start.line][start.col:end.col])
	}
	parts := []string{string(lines[start.line][start.col:])}
	for i := start.line + 1; i < end.line; i++ {
		parts = append(parts, string(lines[i]))
	}
	parts = append(parts, string(lines[end.line][:end.col]))
	return strings.Join(parts, "\n")
}

// delSel 删除选区:首尾行拼接、中间行移除;光标落选区起点
func delSel(b *Buffer, start, end selPos) {
	head := append([]rune{}, b.lines[start.line][:start.col]...)
	tail := append([]rune{}, b.lines[end.line][end.col:]...)
	b.lines[start.line] = append(head, tail...)
	n := end.line - start.line
	if n > 0 {
		copy(b.lines[start.line+1:], b.lines[start.line+1+n:])
		b.lines = b.lines[:len(b.lines)-n]
	}
	b.cur, b.col = start.line, start.col
	b.resetWant()
}
```

```go
package edit

import "strings"

// match 一处查找匹配:逻辑行 + [col,end) rune 区间
type match struct {
	line, col, end int
}

// findMatches 忽略大小写全量扫描;空词返回 nil。
// 用 rune 级 ToLower 折叠(CJK 无影响;个别 Unicode 大写在折叠后 rune 数变化,
// 属可接受近似,md 笔记场景不出现)
func findMatches(lines [][]rune, q string) []match {
	qr := []rune(strings.ToLower(q))
	if len(qr) == 0 {
		return nil
	}
	var out []match
	for li, lr := range lines {
		low := []rune(strings.ToLower(string(lr)))
		for i := 0; i+len(qr) <= len(low); i++ {
			if string(low[i:i+len(qr)]) == string(qr) {
				out = append(out, match{line: li, col: i, end: i + len(qr)})
			}
		}
	}
	return out
}
```

```go
package edit

import "strings"

// leadIndent 行首空白 rune 数(空格/制表各计 1)
func leadIndent(lr []rune) int {
	n := 0
	for _, r := range lr {
		if r != ' ' && r != '\t' {
			break
		}
		n++
	}
	return n
}

// isListItem 是否列表行(无序/有序)
func isListItem(lr []rune) bool {
	s := string(lr)
	return reBullet.MatchString(s) || reOrdered.MatchString(s)
}

// itemSpan 行 i 所属块 [start,end):列表项含缩进更深的连续续行
// (止于空行或同深/更浅行);普通行块=单逻辑行
func itemSpan(lines [][]rune, i int) (int, int) {
	if !isListItem(lines[i]) {
		return i, i + 1
	}
	ind := leadIndent(lines[i])
	end := i + 1
	for end < len(lines) {
		l := lines[end]
		if strings.TrimSpace(string(l)) == "" || leadIndent(l) <= ind {
			break
		}
		end++
	}
	return i, end
}

func concatBlock(parts ...[][]rune) [][]rune {
	n := 0
	for _, p := range parts {
		n += len(p)
	}
	out := make([][]rune, 0, n)
	for _, p := range parts {
		out = append(out, p...)
	}
	return out
}

// moveUpDown 计算块上/下移后的新行表与新光标行;到边界不可移返回 false。
// 有序列表编号不重排(设计决策:原样移动)
func moveUpDown(lines [][]rune, cur int, up bool) ([][]rune, int, bool) {
	s, e := itemSpan(lines, cur)
	var out [][]rune
	newStart := 0
	switch {
	case up && s == 0, !up && e >= len(lines):
		return nil, 0, false
	case up:
		ps, _ := itemSpan(lines, s-1)
		out = concatBlock(lines[:ps], lines[s:e], lines[ps:s], lines[e:])
		newStart = ps
	default:
		_, ne := itemSpan(lines, e)
		out = concatBlock(lines[:s], lines[e:ne], lines[s:e], lines[ne:])
		newStart = ne - (e - s)
	}
	return out, newStart + (cur - s), true
}
```

```go
// MoveLine 上/下移当前行(或列表项块);到边界不动返回 false
func (b *Buffer) MoveLine(up bool) bool {
	out, newCur, ok := moveUpDown(b.lines, b.cur, up)
	if !ok {
		return false
	}
	b.lines = out
	b.cur = newCur
	if b.col > len(b.lines[b.cur]) {
		b.col = len(b.lines[b.cur])
	}
	b.resetWant()
	return true
}

func (b *Buffer) CanMove(up bool) bool {
	_, _, ok := moveUpDown(b.lines, b.cur, up)
	return ok
}
```

```go
// listPrefix 列表行前缀宽度(-␣→2、1.␣→3、10.␣→4);非列表行 0
func listPrefix(s string) int {
	if m := reBullet.FindStringSubmatch(s); m != nil {
		return len([]rune(m[1] + m[2] + " "))
	}
	if m := reOrdered.FindStringSubmatch(s); m != nil {
		return len([]rune(m[1] + m[2] + m[3] + " "))
	}
	return 0
}

// Indent 列表行按前缀宽度缩进(嵌套层级符合 CommonMark),非列表行插 2 空格
func (b *Buffer) Indent() {
	if w := listPrefix(string(b.lines[b.cur])); w > 0 {
		pad := make([]rune, w)
		for i := range pad {
			pad[i] = ' '
		}
		b.lines[b.cur] = append(pad, b.lines[b.cur]...)
		b.col += w
		b.resetWant()
		return
	}
	b.Insert(' ', ' ')
}

// Outdent 反缩进:删行首至多一个前缀宽度(非列表行至多 2)的空格,下限 0
func (b *Buffer) Outdent() {
	w := 2
	if p := listPrefix(string(b.lines[b.cur])); p > 0 {
		w = p
	}
	lr := b.lines[b.cur]
	n := leadIndent(lr)
	if n > w {
		n = w
	}
	if n > 0 {
		b.lines[b.cur] = append([]rune{}, lr[n:]...)
		if b.col < n {
			b.col = 0
		} else {
			b.col -= n
		}
	}
	b.resetWant()
}

// CanOutdent 行首有空白可反缩进(供命令层先判后压撤销栈)
func (b *Buffer) CanOutdent() bool { return leadIndent(b.lines[b.cur]) > 0 }
```

```go
// selSpan 本显示行上选区覆盖区间 [lo,hi);无覆盖 ok=false
func selSpan(r VRow, s, t selPos) (int, int, bool) {
	if r.Line < s.line || r.Line > t.line {
		return 0, 0, false
	}
	lo, hi := r.Start, r.End
	if r.Line == s.line && s.col > lo {
		lo = s.col
	}
	if r.Line == t.line && t.col < hi {
		hi = t.col
	}
	return lo, hi, hi > lo
}

// rowStyled 光标行/选区行渲染:选区暗底,光标格反显最优先(含行尾/幻影格)
func (e editor) rowStyled(r VRow) string {
	_, col := e.buf.Cur()
	lr := e.buf.LineRunes(r.Line)
	var b strings.Builder
	if DispCol(lr, r.Start) > 0 {
		b.WriteString(wrapIndent)
	}
	selLo, selHi := -1, -1
	if s, t, ok := e.selRange(); ok {
		if a, c, hit := selSpan(r, s, t); hit {
			selLo, selHi = a, c
		}
	}
	for i := r.Start; i <= r.End; i++ {
		switch {
		case i == col:
			if col < r.End {
				b.WriteString(cursorStyle.Render(string(lr[col])))
			} else {
				b.WriteString(cursorStyle.Render(" "))
			}
		case i >= selLo && i < selHi:
			j := selHi
			if j > r.End {
				j = r.End
			}
			b.WriteString(selStyle.Render(string(lr[i:j])))
			i = j - 1
		default:
			if i < r.End {
				b.WriteString(string(lr[i]))
			}
		}
	}
	return b.String()
}
```

```go
func (e *editor) recalcMatches() {
	e.matches = findMatches(e.buf.lines, e.q)
	e.mi = 0
	line, col := e.buf.Cur()
	for i, m := range e.matches {
		if m.line > line || (m.line == line && m.col >= col) {
			e.mi = i
			return
		}
	}
}

func (e *editor) gotoMatch(i int) {
	m := e.matches[i]
	e.buf.cur, e.buf.col = m.line, m.col
	e.buf.resetWant()
	e.followCursor()
}

// jumpMatch 沿方向跳相邻匹配(回绕);无匹配不动
func (e *editor) jumpMatch(d int) {
	n := len(e.matches)
	if n == 0 {
		return
	}
	e.mi = ((e.mi+d)%n + n) % n
	e.gotoMatch(e.mi)
}

// jumpNextMatch 跳光标之后首个匹配,无则回绕首个
func (e *editor) jumpNextMatch() {
	if len(e.matches) == 0 {
		return
	}
	line, col := e.buf.Cur()
	for i, m := range e.matches {
		if m.line > line || (m.line == line && m.col > col) {
			e.mi = i
			e.gotoMatch(i)
			return
		}
	}
	e.mi = 0
	e.gotoMatch(0)
}
```

```go
// rowStyled 渲染需上样式的一行:查找匹配(当前匹配反显)/选区暗底,光标格最优先
func (e editor) rowStyled(r VRow) string {
	_, col := e.buf.Cur()
	lr := e.buf.LineRunes(r.Line)
	var spans []span
	if s, t, ok := e.selRange(); ok {
		if a, c, hit := selSpan(r, s, t); hit {
			spans = append(spans, span{a, c, selStyle})
		}
	}
	for i, m := range e.matches {
		if m.line != r.Line || m.end <= r.Start || m.col >= r.End {
			continue
		}
		lo, hi := m.col, m.end
		if lo < r.Start {
			lo = r.Start
		}
		if hi > r.End {
			hi = r.End
		}
		st := matchStyle
		if i == e.mi {
			st = cursorStyle
		}
		spans = append(spans, span{lo, hi, st})
	}
	var b strings.Builder
	if DispCol(lr, r.Start) > 0 {
		b.WriteString(wrapIndent)
	}
	for i := r.Start; i <= r.End; i++ {
		if i == col {
			if col < r.End {
				b.WriteString(cursorStyle.Render(string(lr[col])))
			} else {
				b.WriteString(cursorStyle.Render(" "))
			}
			continue
		}
		s := ""
		if i < r.End {
			s = string(lr[i])
		}
		for _, sp := range spans { // 后入者覆盖:选区先入,匹配后入
			if i >= sp.lo && i < sp.hi {
				s = sp.st.Render(string(lr[i]))
			}
		}
		b.WriteString(s)
	}
	return b.String()
}
```
