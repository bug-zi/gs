```go
package command

import (
	"io/fs"
	"path/filepath"
	"strings"

	"github.com/bugzi/gs/internal/workspace"
)

// ListKnowledge 返回三层知识库（sea/river/spring）下全部 .md 的工作区相对路径（正斜杠）。
// 目录不存在视为空；错误向上返回。
func ListKnowledge(ws *workspace.Workspace) ([]string, error) {
	var out []string
	dirs := []string{ws.SeaDir(), ws.RiverDir(), ws.SpringDir()}
	for _, dir := range dirs {
		err := filepath.WalkDir(dir, func(path string, d fs.DirEntry, err error) error {
			if err != nil {
				return nil // 单层缺失/不可读不致命
			}
			if d.IsDir() || !strings.EqualFold(filepath.Ext(d.Name()), ".md") {
				return nil
			}
			rel, err := filepath.Rel(ws.Root, path)
			if err != nil {
				return nil
			}
			out = append(out, filepath.ToSlash(rel))
			return nil
		})
		if err != nil {
			return nil, err
		}
	}
	return out, nil
}
```

```go
import (
```

```go
// ListLegacyNotes 列出遗留 notes 目录下全部 .md 的相对路径（相对 notes 目录，正斜杠）。
// 目录不存在返回空。
func ListLegacyNotes(ws *workspace.Workspace) ([]string, error) {
	var names []string
	root := ws.NotesDir()
	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // 目录不存在等情况
		}
		if d.IsDir() || !strings.EqualFold(filepath.Ext(d.Name()), ".md") {
			return nil
		}
		rel, err := filepath.Rel(root, path)
		if err != nil {
			return nil
		}
		names = append(names, filepath.ToSlash(rel))
		return nil
	})
	return names, err
}

// MoveNote 把遗留笔记迁入指定层（"sea"|"river"），保留相对子路径，返回新位置。
func MoveNote(ws *workspace.Workspace, name, tier string) (string, error) {
	if tier != "sea" && tier != "river" {
		return "", fmt.Errorf("未知层: %s", tier)
	}
	dir := ws.RiverDir()
	if tier == "sea" {
		dir = ws.SeaDir()
	}
	dst := filepath.Join(dir, filepath.FromSlash(name))
	if _, err := os.Stat(dst); err == nil {
		return "", fmt.Errorf("目标已存在: %s", ws.Rel(dst))
	}
	if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
		return "", err
	}
	src := filepath.Join(ws.NotesDir(), filepath.FromSlash(name))
	if err := os.Rename(src, dst); err != nil {
		return "", err
	}
	return ws.Rel(dst), nil
}

// legacyInterview 一个待收集的面试题文件
type legacyInterview struct {
	Path    string // 工作区相对路径（正斜杠）
	Project string // 所属项目目录名
	Target  string // 迁入 sea/interview 后的文件名（项目名前缀防撞）
}

// ListLegacyInterview 扫描 Study/Project 与 Project 下路径含 interview/面试 的 md。
func ListLegacyInterview(ws *workspace.Workspace) ([]legacyInterview, error) {
	var out []legacyInterview
	for _, base := range []string{filepath.Join("Study", "Project"), "Project"} {
		root := filepath.Join(ws.Root, base)
		_ = filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
			if err != nil || d.IsDir() {
				return nil
			}
			if !strings.EqualFold(filepath.Ext(d.Name()), ".md") {
				return nil
			}
			rel, err := filepath.Rel(ws.Root, path)
			if err != nil {
				return nil
			}
			rs := filepath.ToSlash(rel)
			lp := strings.ToLower(rs)
			if !strings.Contains(lp, "interview") && !strings.Contains(lp, "面试") {
				return nil
			}
			relBase, err := filepath.Rel(root, path)
			if err != nil {
				return nil
			}
			segs := strings.Split(filepath.ToSlash(relBase), "/")
			project := segs[0]
			out = append(out, legacyInterview{
				Path:    rs,
				Project: project,
				Target:  project + "-" + d.Name(),
			})
			return nil
		})
	}
	return out, nil
}

// MoveInterview 把面试题文件迁入 sea/interview/，返回新位置；同名目标已存在则报错。
func MoveInterview(ws *workspace.Workspace, li legacyInterview) (string, error) {
	dst := filepath.Join(ws.SeaDir(), "interview", filepath.FromSlash(li.Target))
	if _, err := os.Stat(dst); err == nil {
		return "", fmt.Errorf("目标已存在: %s", ws.Rel(dst))
	}
	if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
		return "", err
	}
	src := filepath.Join(ws.Root, filepath.FromSlash(li.Path))
	if err := os.Rename(src, dst); err != nil {
		return "", err
	}
	return ws.Rel(dst), nil
}
```

```go
// previewOf 读文件前 n 行作预览
func previewOf(path string, n int) string {
	data, err := os.ReadFile(path)
	if err != nil {
		return "（读取失败）"
	}
	lines := strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n")
	if len(lines) > n {
		lines = lines[:n]
	}
	return strings.Join(lines, "\n")
}

// runMigrate 交互迁移：遗留笔记逐个人工判断归层，面试题逐个确认收集
func runMigrate(ctx *Ctx) error {
	for _, d := range []string{ctx.WS.SeaDir(), ctx.WS.RiverDir(), ctx.WS.SpringDir()} {
		if err := os.MkdirAll(d, 0o755); err != nil {
			return err
		}
	}
	var moved, skipped int
	names, err := ListLegacyNotes(ctx.WS)
	if err != nil {
		return err
	}
	for _, n := range names {
		fmt.Fprintf(ctx.Out, "\n%s\n%s\n", ui.Path("Study/notes/"+n), previewOf(filepath.Join(ctx.WS.NotesDir(), filepath.FromSlash(n)), 10))
		var tier string
		opts := []huhOption{
			option("河流 river（已筛选整理过）", "river"),
			option("大海 sea（粘贴原样）", "sea"),
			option("跳过（留在原处）", ""),
		}
		if err := selectFrom(ctx, "归入哪一层？", opts, &tier); err != nil {
			return err
		}
		if tier == "" {
			skipped++
			continue
		}
		dst, err := MoveNote(ctx.WS, n, tier)
		if err != nil {
			fmt.Fprintln(ctx.Out, ui.Dim("跳过: "+err.Error()))
			skipped++
			continue
		}
		fmt.Fprintf(ctx.Out, "已迁入 %s\n", ui.Path(dst))
		moved++
	}
	lis, err := ListLegacyInterview(ctx.WS)
	if err != nil {
		return err
	}
	var iMoved int
	for _, li := range lis {
		fmt.Fprintf(ctx.Out, "\n%s（项目 %s）\n%s\n", ui.Path(li.Path), li.Project, previewOf(filepath.Join(ctx.WS.Root, filepath.FromSlash(li.Path)), 10))
		ok, err := confirm(ctx, "迁入 sea/interview？（将命名为 "+li.Target+"）")
		if err != nil {
			return err
		}
		if !ok {
			skipped++
			continue
		}
		dst, err := MoveInterview(ctx.WS, li)
		if err != nil {
			fmt.Fprintln(ctx.Out, ui.Dim("跳过: "+err.Error()))
			skipped++
			continue
		}
		fmt.Fprintf(ctx.Out, "已迁入 %s\n", ui.Path(dst))
		iMoved++
	}
	fmt.Fprintf(ctx.Out, "\n迁移完成：笔记 %d · 面试题 %d · 跳过 %d\n", moved, iMoved, skipped)
	if moved+iMoved == 0 && skipped == 0 {
		fmt.Fprintln(ctx.Out, "没有发现需要迁移的遗留内容")
	}
	if rest, _ := ListLegacyNotes(ctx.WS); len(rest) == 0 {
		fmt.Fprintln(ctx.Out, ui.Dim("notes 目录已清空，可自行删除 Study/notes 目录"))
	}
	return nil
}

		Run: func(ctx *Ctx) error {
			return runMigrate(ctx)
		},
	}
}
```
