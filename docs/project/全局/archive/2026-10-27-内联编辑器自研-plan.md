```go
package edit

import "strings"

// Buffer 多行文本缓冲;光标以 (逻辑行下标, 行内 rune 下标) 表示。
// want 为 ↑↓ 移动时的粘滞视觉列,-1 表示未激活。
type Buffer struct {
	lines [][]rune
	cur   int
	col   int
	want  int
}

func NewBuffer(s string) *Buffer {
	b := &Buffer{want: -1}
	for _, l := range strings.Split(s, "\n") {
		b.lines = append(b.lines, []rune(l))
	}
	return b
}

func (b *Buffer) Value() string {
	parts := make([]string, len(b.lines))
	for i, l := range b.lines {
		parts[i] = string(l)
	}
	return strings.Join(parts, "\n")
}

func (b *Buffer) LineCount() int         { return len(b.lines) }
func (b *Buffer) Cur() (line, col int)   { return b.cur, b.col }
func (b *Buffer) LineRunes(i int) []rune { return b.lines[i] }

func (b *Buffer) resetWant() { b.want = -1 }

// Insert 在光标处插入字符
func (b *Buffer) Insert(rs ...rune) {
	l := append(b.lines[b.cur], make([]rune, len(rs))...)
	copy(l[b.col+len(rs):], l[b.col:])
	copy(l[b.col:], rs)
	b.lines[b.cur] = l
	b.col += len(rs)
	b.resetWant()
}

// Backspace 删除光标前一字符;行首则与上一行合并
func (b *Buffer) Backspace() {
	if b.col > 0 {
		b.lines[b.cur] = append(b.lines[b.cur][:b.col-1], b.lines[b.cur][b.col:]...)
		b.col--
		b.resetWant()
		return
	}
	if b.cur > 0 {
		prev := b.lines[b.cur-1]
		b.lines[b.cur-1] = append(prev, b.lines[b.cur]...)
		b.lines = append(b.lines[:b.cur], b.lines[b.cur+1:]...)
		b.cur--
		b.col = len(prev)
		b.resetWant()
	}
}

// Delete 删除光标处字符;行尾则吞并下一行
func (b *Buffer) Delete() {
	l := b.lines[b.cur]
	if b.col < len(l) {
		b.lines[b.cur] = append(l[:b.col], l[b.col+1:]...)
		b.resetWant()
		return
	}
	if b.cur < len(b.lines)-1 {
		b.lines[b.cur] = append(l, b.lines[b.cur+1]...)
		b.lines = append(b.lines[:b.cur+1], b.lines[b.cur+2:]...)
		b.resetWant()
	}
}

// Break 在光标处断行,不带任何前缀(Enter 的列表续写建立在它之上;粘贴换行直接用它)
func (b *Buffer) Break() {
	l := b.lines[b.cur]
	head := append([]rune{}, l[:b.col]...)
	rest := append([]rune{}, l[b.col:]...)
	b.lines[b.cur] = head
	b.lines = append(b.lines[:b.cur+1], append([][]rune{rest}, b.lines[b.cur+1:]...)...)
	b.cur++
	b.col = 0
	b.resetWant()
}

func (b *Buffer) Left() {
	if b.col > 0 {
		b.col--
	} else if b.cur > 0 {
		b.cur--
		b.col = len(b.lines[b.cur])
	}
	b.resetWant()
}

func (b *Buffer) Right() {
	if b.col < len(b.lines[b.cur]) {
		b.col++
	} else if b.cur < len(b.lines)-1 {
		b.cur++
		b.col = 0
	}
	b.resetWant()
}

func (b *Buffer) Top() {
	b.cur, b.col = 0, 0
	b.resetWant()
}

func (b *Buffer) Bottom() {
	b.cur = len(b.lines) - 1
	b.col = len(b.lines[b.cur])
	b.resetWant()
}
```

```go
var (
	reBullet      = regexp.MustCompile(`^([ \t]*)([-*+])[ \t]+`)
	reOrdered     = regexp.MustCompile(`^([ \t]*)(\d+)([.)])[ \t]+`)
	reBulletOnly  = regexp.MustCompile(`^([ \t]*)([-*+])[ \t]*$`)
	reOrderedOnly = regexp.MustCompile(`^([ \t]*)(\d+)([.)])[ \t]*$`)
)

// Enter 换行:列表行(无序/有序)自动续写前缀,有序编号递增;
// 空列表项(只有前缀无内容)回车清除前缀退出列表,不插行
func (b *Buffer) Enter() {
	l := b.lines[b.cur]
	head := string(l[:b.col])
	tail := string(l[b.col:])
	prefix := ""
	switch {
	case tail == "" && reBulletOnly.MatchString(head):
		m := reBulletOnly.FindStringSubmatch(head)
		b.lines[b.cur] = []rune(m[1])
		b.col = len(m[1])
		b.resetWant()
		return
	case tail == "" && reOrderedOnly.MatchString(head):
		m := reOrderedOnly.FindStringSubmatch(head)
		b.lines[b.cur] = []rune(m[1])
		b.col = len(m[1])
		b.resetWant()
		return
	case reBullet.MatchString(head):
		m := reBullet.FindStringSubmatch(head)
		prefix = m[1] + m[2] + " "
	case reOrdered.MatchString(head):
		m := reOrdered.FindStringSubmatch(head)
		n, _ := strconv.Atoi(m[2])
		prefix = m[1] + strconv.Itoa(n+1) + m[3] + " "
	}
	b.Break() // 光标归新行首
	if prefix != "" {
		b.Insert([]rune(prefix)...)
	}
}
```

```go
package edit

import rw "github.com/mattn/go-runewidth"

// Segment 是一行按显示宽度切出的显示片段
type Segment struct {
	Runes   []rune
	Wrapped bool // true = 软换行延续片段(渲染加缩进标记)
}

// WrapRunes 把 line 按显示宽度 width 切成显示片段。
// 纯字符切分(不做断词),保证:每片段显示宽 ≤ width;
// 空行返回单个未换行片段;整行恰填满 width 时末尾补空片段给光标落脚。
func WrapRunes(line []rune, width int) []Segment {
	if width < 1 {
		width = 1
	}
	var segs []Segment
	w, start := 0, 0
	for i, r := range line {
		cw := rw.RuneWidth(r)
		if w+cw > width {
			segs = append(segs, Segment{Runes: line[start:i]})
			start = i
			w = 0
		}
		w += cw
	}
	segs = append(segs, Segment{Runes: line[start:]})
	if w == width {
		segs = append(segs, Segment{})
	}
	for i := range segs {
		segs[i].Wrapped = i > 0
	}
	return segs
}
```

```go
// VRow 描述一个显示行:逻辑行 Line 的 rune 区间 [Start,End);
// Start>0 即软换行延续片段(渲染加缩进)
type VRow struct {
	Line  int
	Start int
	End   int
}

// Layout 计算整个缓冲在 width 下的显示行布局
func Layout(b *Buffer, width int) []VRow {
	var rows []VRow
	for i := 0; i < b.LineCount(); i++ {
		lr := b.LineRunes(i)
		off := 0
		for _, s := range WrapRunes(lr, width) {
			rows = append(rows, VRow{Line: i, Start: off, End: off + len(s.Runes)})
			off += len(s.Runes)
		}
	}
	return rows
}

// CursorVRow 光标所在显示行序号(全文档显示行坐标系)
func CursorVRow(b *Buffer, width int) int {
	rows := Layout(b, width)
	line, col := b.Cur()
	first, last, hit := -1, -1, -1
	for i, r := range rows {
		if r.Line != line {
			continue
		}
		if first < 0 {
			first = i
		}
		last = i
		if col >= r.Start && col < r.End {
			hit = i
		}
	}
	if hit >= 0 {
		return hit
	}
	if last >= 0 {
		return last // col 在行尾/空行/幻影行 → 该逻辑行最后一个显示行
	}
	return 0
}

// DispCol rune 下标 → 行内显示列
func DispCol(line []rune, idx int) int {
	return rw.StringWidth(string(line[:idx]))
}

// idxAtDispCol 行内显示列 → rune 下标(不超过该列的最大下标)
func idxAtDispCol(line []rune, dcol int) int {
	w := 0
	for i, r := range line {
		cw := rw.RuneWidth(r)
		if w+cw > dcol {
			return i
		}
		w += cw
	}
	return len(line)
}
```

```go
// Up 光标上移一个显示行(软换行片段也算一行),视觉列粘滞
func (b *Buffer) Up(width int) {
	rows := Layout(b, width)
	i := CursorVRow(b, width)
	if i == 0 {
		b.col = 0
		b.resetWant()
		return
	}
	b.step(rows, i, -1)
}

// Down 光标下移一个显示行;已是最后一显示行则到逻辑行尾
func (b *Buffer) Down(width int) {
	rows := Layout(b, width)
	i := CursorVRow(b, width)
	if i >= len(rows)-1 {
		b.col = len(b.lines[b.cur])
		b.resetWant()
		return
	}
	b.step(rows, i, +1)
}

// step 沿显示行 ±1 移动,保持视觉列(片段内显示偏移)粘滞
func (b *Buffer) step(rows []VRow, i, d int) {
	if b.want < 0 {
		cur := rows[i]
		b.want = DispCol(b.lines[b.cur], b.col) - DispCol(b.lines[cur.Line], cur.Start)
	}
	t := rows[i+d]
	lr := b.lines[t.Line]
	base := DispCol(lr, t.Start)
	col := idxAtDispCol(lr, base+b.want)
	if col > t.End {
		col = t.End
	}
	if col < t.Start {
		col = t.Start
	}
	b.cur, b.col = t.Line, col
}

// Home 移到当前显示行首(片段行首即逻辑行中段)
func (b *Buffer) Home(width int) {
	rows := Layout(b, width)
	r := rows[CursorVRow(b, width)]
	b.cur, b.col = r.Line, r.Start
	b.resetWant()
}

// End 移到当前显示行尾(含幻影空片段行)
func (b *Buffer) End(width int) {
	rows := Layout(b, width)
	r := rows[CursorVRow(b, width)]
	b.cur, b.col = r.Line, r.End
	b.resetWant()
}
```

```go
package edit

import (
	"bytes"
	"fmt"
	"strings"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/charmbracelet/lipgloss"

	"github.com/bugzi/gs/internal/ui"
)

const wrapIndent = "  " // 软换行延续片段缩进标记

var cursorStyle = lipgloss.NewStyle().Reverse(true)

// editor 内联编辑 tea 模型;buffer/wrap 为纯逻辑,本文件只做按键分流、滚动与渲染
type editor struct {
	buf     *Buffer
	path    string
	crlf    bool
	initial string
	width   int
	height  int // 编辑区可见显示行数
	offset  int // 顶部显示行序号(全文档显示行坐标系)
	err     error
	confirm bool
}

func newEditor(path string, data []byte) editor {
	e := editor{
		buf:     NewBuffer(normalizeCRLF(data)),
		path:    path,
		crlf:    bytes.Contains(data, []byte("\r\n")),
		initial: normalizeCRLF(data),
	}
	e.buf.Bottom() // 打开既有文件时光标落在文档末,续写体验与旧 textarea 一致
	return e
}

func (e editor) Init() tea.Cmd { return nil }

func (e editor) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		e.width = msg.Width
		e.height = msg.Height - 3 // 标题行 + 空行 + 确认/错误行余量
		if e.height < 1 {
			e.height = 1
		}
		e.clampOffset()
		return e, nil
	case tea.PasteMsg:
		e.paste(string(msg))
		e.followCursor()
		return e, nil
	case tea.KeyMsg:
		switch msg.Type {
		case tea.KeyCtrlS:
			if err := saveFile(e.path, e.buf.Value(), e.crlf); err != nil {
				e.err = err
				return e, nil
			}
			return e, tea.Quit
		case tea.KeyEsc:
			if e.confirm {
				e.confirm = false
				return e, nil
			}
			if e.buf.Value() != e.initial {
				e.confirm = true
				return e, nil
			}
			return e, tea.Quit
		}
		if e.confirm { // 确认态吞掉其余按键
			switch strings.ToLower(msg.String()) {
			case "y":
				return e, tea.Quit
			case "n":
				e.confirm = false
			}
			return e, nil
		}
		switch msg.Type {
		case tea.KeyRunes, tea.KeySpace:
			rs := msg.Runes
			if msg.Type == tea.KeySpace && len(rs) == 0 {
				rs = []rune{' '}
			}
			e.buf.Insert(rs...)
		case tea.KeyEnter:
			e.buf.Enter()
		case tea.KeyBackspace:
			e.buf.Backspace()
		case tea.KeyDelete:
			e.buf.Delete()
		case tea.KeyLeft:
			e.buf.Left()
		case tea.KeyRight:
			e.buf.Right()
		case tea.KeyUp:
			e.buf.Up(e.width)
		case tea.KeyDown:
			e.buf.Down(e.width)
		case tea.KeyHome:
			e.buf.Home(e.width)
		case tea.KeyEnd:
			e.buf.End(e.width)
		case tea.KeyPgUp:
			e.offset -= e.height
			e.clampOffset()
			return e, nil
		case tea.KeyPgDn:
			e.offset += e.height
			e.clampOffset()
			return e, nil
		default:
			return e, nil
		}
		e.followCursor()
		return e, nil
	}
	return e, nil
}

// paste 粘贴文本;换行用 Break,不走列表续写
func (e *editor) paste(s string) {
	for i, p := range strings.Split(normalizeCRLF([]byte(s)), "\n") {
		if i > 0 {
			e.buf.Break()
		}
		if p != "" {
			e.buf.Insert([]rune(p)...)
		}
	}
}

// clampOffset 把 offset 夹回合法范围
func (e *editor) clampOffset() {
	if e.width < 1 || e.height < 1 {
		return
	}
	n := e.rowCount()
	if n < e.height {
		n = e.height
	}
	if e.offset > n-e.height {
		e.offset = n - e.height
	}
	if e.offset < 0 {
		e.offset = 0
	}
}

func (e *editor) rowCount() int { return len(Layout(e.buf, e.width)) }

// followCursor 滚动保证光标显示行可见
func (e *editor) followCursor() {
	if e.width < 1 || e.height < 1 {
		return
	}
	i := CursorVRow(e.buf, e.width)
	if i < e.offset {
		e.offset = i
	} else if i >= e.offset+e.height {
		e.offset = i - e.height + 1
	}
}

func (e editor) View() string {
	var b strings.Builder
	line, col := e.buf.Cur()
	mod := ""
	if e.buf.Value() != e.initial {
		mod = " · 已修改"
	}
	fmt.Fprintf(&b, "编辑 %s · 行 %d,列 %d · %s · %s%s\n\n",
		ui.Path(e.path), line+1, DispCol(e.buf.LineRunes(line), col)+1,
		ui.Dim("Ctrl+S 保存退出"), ui.Dim("Esc 退出"), ui.Dim(mod))
	rows := Layout(e.buf, e.width)
	curRow := CursorVRow(e.buf, e.width)
	end := e.offset + e.height
	if end > len(rows) {
		end = len(rows)
	}
	if e.offset < len(rows) {
		for i := e.offset; i < end; i++ {
			r := rows[i]
			if i == curRow {
				b.WriteString(e.cursorRowStr(r))
			} else {
				b.WriteString(e.rowText(r))
			}
			b.WriteByte('\n')
		}
	}
	if e.confirm {
		b.WriteString("\n有未保存修改，放弃并退出？(y/N)")
	}
	if e.err != nil {
		fmt.Fprintf(&b, "\n保存失败: %v", e.err)
	}
	return b.String()
}

// rowText 普通显示行;延续片段(逻辑行中段起)加缩进标记
func (e editor) rowText(r VRow) string {
	lr := e.buf.LineRunes(r.Line)
	s := string(lr[r.Start:r.End])
	if DispCol(lr, r.Start) > 0 {
		s = wrapIndent + s
	}
	return s
}

// cursorRowStr 光标所在显示行:光标字符反显(行尾/幻影行为反显空格)
func (e editor) cursorRowStr(r VRow) string {
	line, col := e.buf.Cur()
	lr := e.buf.LineRunes(line)
	s := ""
	if DispCol(lr, r.Start) > 0 {
		s = wrapIndent
	}
	if col < r.Start || col > r.End {
		return s + string(lr[r.Start:r.End])
	}
	cur := " "
	if col < r.End {
		cur = string(lr[col])
	}
	return s + string(lr[r.Start:col]) + cursorStyle.Render(cur) + string(lr[col+1:r.End])
}
```

```go
package edit

import (
	"bytes"
	"strings"

	tea "github.com/charmbracelet/bubbletea"

	"github.com/bugzi/gs/internal/workspace"
)

func normalizeCRLF(data []byte) string { return strings.ReplaceAll(string(data), "\r\n", "\n") }

// saveFile 按 detect 出的换行风格原子写回(项目纪律:一律原子写)
func saveFile(path, content string, crlf bool) error {
	out := []byte(content)
	if crlf {
		out = bytes.ReplaceAll(out, []byte("\n"), []byte("\r\n"))
	}
	return workspace.WriteFileAtomic(path, out)
}

// EditInline CLI 全屏编辑 path;tea 启动失败返回错误,由调用方降级外部编辑器
func EditInline(path string, data []byte) error {
	m := newEditor(path, data)
	p, err := tea.NewProgram(m, tea.WithAltScreen(), tea.WithBracketedPaste()).Run()
	if err != nil {
		return err
	}
	if fin, ok := p.(editor); ok {
		return fin.err // 从最终模型取保存错误(旧实现读初始副本,错误提示永不显示)
	}
	return nil
}
```
