```go
// shippedSeaSet 已装船基名集合：ship 同名副本（加工中）∪ river 去编号名（已入河）。
// 层目录列举失败时静默跳过该层——宁漏标、不阻塞浏览。
func shippedSeaSet(ws *workspace.Workspace) map[string]bool {
	set := map[string]bool{}
	if files, err := ListTierFiles(ws, "ship"); err == nil {
		for _, f := range files {
			set[strings.TrimSuffix(filepath.Base(f), ".md")] = true
		}
	}
	if files, err := ListTierFiles(ws, "river"); err == nil {
		for _, f := range files {
			set[roadmap.StripNumPrefix(filepath.Base(f))] = true
		}
	}
	return set
}
```

```go
// buildDirGroups 按一级子目录聚合（sea）；根目录散件归 looseName 组，组与组内条目均按名排序。
// shipped 为已装船基名集合（nil 合法）：命中的条目带 Shipped 标记，组内未装船排前、已装船排后。
func buildDirGroups(tier string, files []string, looseName string, shipped map[string]bool) []TierGroup {
	prefix := "Study/" + tier + "/"
	byDir := map[string][]TierItem{}
	var loose []TierItem
	for _, f := range files {
		rel := strings.TrimPrefix(f, prefix)
		name := strings.TrimSuffix(filepath.Base(rel), ".md")
		it := TierItem{Title: name, Path: f, Shipped: shipped[name]}
		if i := strings.Index(rel, "/"); i >= 0 {
			dir := rel[:i]
			byDir[dir] = append(byDir[dir], it)
		} else {
			loose = append(loose, it)
		}
	}
	var groups []TierGroup
	for dir, items := range byDir {
		sortShippedFirst(items)
		groups = append(groups, TierGroup{Name: dir, Items: items})
	}
	sort.Slice(groups, func(i, j int) bool { return groups[i].Name < groups[j].Name })
	sortShippedFirst(loose)
	if len(loose) > 0 {
		groups = append(groups, TierGroup{Name: looseName, Items: loose})
	}
	return groups
}

// sortShippedFirst 组内排序：未装船在前，各自按名升序
func sortShippedFirst(items []TierItem) {
	sort.Slice(items, func(i, j int) bool {
		if items[i].Shipped != items[j].Shipped {
			return !items[i].Shipped
		}
		return items[i].Title < items[j].Title
	})
}
```

```go
// plainOptionsText 非交互降级编号列表文本；Dim 项同样套阴影色
func plainOptionsText(title string, opts []menuOption) string {
	var b strings.Builder
	b.WriteString(title + "\n")
	for i, o := range opts {
		label := o.Label
		if o.Dim {
			label = ui.Dim(label)
		}
		fmt.Fprintf(&b, "  %d. %s\n", i+1, label)
	}
	b.WriteString("输入编号选择（空/0 取消）> ")
	return b.String()
}

// plainSelect 非交互降级：编号列表 + 读行选择
func plainSelect(ctx *Ctx, title string, opts []menuOption, out *string) error {
	fmt.Fprint(ctx.Out, plainOptionsText(title, opts))
	line, err := bufio.NewReader(os.Stdin).ReadString('\n')
	if err != nil {
		return ErrCancelled
	}
	n, perr := strconv.Atoi(strings.TrimSpace(line))
	if perr != nil || n < 1 || n > len(opts) {
		return ErrCancelled
	}
	*out = opts[n-1].Value
	return nil
}
```
