// tierDirs 层名 → 目录
var tierDirs = map[string]func(*workspace.Workspace) string{
	"sea":    (*workspace.Workspace).SeaDir,
	"river":  (*workspace.Workspace).RiverDir,
	"spring": (*workspace.Workspace).SpringDir,
	"code":   (*workspace.Workspace).CodeDir,
}

// tierExts 层名 → 条目扩展名（code 层是 .go 演示代码，其余层为 .md 笔记）
var tierExts = map[string]string{
	"sea": ".md", "river": ".md", "spring": ".md", "code": ".go",
}

// ListTierFiles 返回指定层（sea/river/spring/code）下全部条目的工作区相对路径（正斜杠），
// 按层过滤扩展名（tierExts）
func ListTierFiles(ws *workspace.Workspace, tier string) ([]string, error) {
	dirFn, ok := tierDirs[tier]
	if !ok {
		return nil, errors.New("未知层: " + tier)
	}
	ext := tierExts[tier]
	var out []string
	err := filepath.WalkDir(dirFn(ws), func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // 层目录缺失视为空
		}
		if d.IsDir() || !strings.EqualFold(filepath.Ext(d.Name()), ext) {
			return nil
		}
		rel, err := filepath.Rel(ws.Root, path)
		if err != nil {
			return nil
		}
		out = append(out, filepath.ToSlash(rel))
		return nil
	})
	return out, err
}
```

注意：`ListKnowledge` 保持 sea→river→spring 三层不变（spec 非目标：搜索不纳入 .go）。

// CodeTemplate 知识点演示代码初始模板：可运行、gofmt 合规
func CodeTemplate(topic, rel string) string {
	return fmt.Sprintf("// 知识点「%s」演示代码\n// 运行：go run %q\n\npackage main\n\nimport \"fmt\"\n\nfunc main() {\n\t// TODO: 写下与该知识点相关的演示代码\n\tfmt.Println(%q)\n}\n", topic, rel, topic)
}

// CreateCodeEntry 创建知识点演示代码 Study/code/<标题>.go，返回工作区相对路径，原子写
func CreateCodeEntry(ws *workspace.Workspace, topic string) (string, error) {
	if err := validateTopic(topic); err != nil {
		return "", err
	}
	path := filepath.Join(ws.CodeDir(), topic+".go")
	if _, err := os.Stat(path); err == nil {
		return "", fmt.Errorf("条目已存在: %s", ws.Rel(path))
	}
	if err := os.MkdirAll(ws.CodeDir(), 0o755); err != nil {
		return "", err
	}
	if err := workspace.WriteFileAtomic(path, []byte(CodeTemplate(topic, ws.Rel(path)))); err != nil {
		return "", err
	}
	return ws.Rel(path), nil
}
```

（`fmt.Sprintf("%q", rel)` 对中文/正斜杠保留字面，生成 `go run "Study/code/标题.go"` 可直接复制运行。）

// runCodeAdd 创建/打开知识点唯一的演示代码并绑定（镜像 runSpringAdd）：
// 定位知识点 → Study/code/<标题>.go 已绑定则直接打开；否则创建模板文件并 BindCode 落盘
func runCodeAdd(ctx *Ctx, kw string) error {
	rm, err := loadRoadmap(ctx)
	if err != nil {
		return err
	}
	ref, err := locatePoint(ctx, rm, kw)
	if err != nil {
		return err
	}
	p := rm.PointAt(ref)
	if p.Code != "" {
		abs := filepath.Join(ctx.WS.Root, filepath.FromSlash(p.Code))
		fmt.Fprintf(ctx.Out, "已存在 %s，直接打开\n", ui.Path(p.Code))
		return editOpen(ctx, abs)
	}
	rel, err := CreateCodeEntry(ctx.WS, p.Title)
	if err != nil {
		return err
	}
	rm.BindCode(ref, rel)
	if err := roadmap.Save(ctx.WS.RoadmapFile(), rm); err != nil {
		return err
	}
	fmt.Fprintf(ctx.Out, "已创建并绑定 %s\n", ui.Path(rel))
	return editOpen(ctx, filepath.Join(ctx.WS.Root, filepath.FromSlash(rel)))
}
