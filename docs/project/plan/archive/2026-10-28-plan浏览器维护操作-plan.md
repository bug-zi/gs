```go
func (rm *Roadmap) UnbindSpring(r PointRef) { rm.PointAt(r).Spring = "" }

// UnbindCode 清空 code 绑定
func (rm *Roadmap) UnbindCode(r PointRef) { rm.PointAt(r).Code = "" }

// BindShip 追加绑定（幂等）
func (rm *Roadmap) BindShip(r PointRef, rel string) { bindList(&rm.PointAt(r).Ship, rel) }

// UnbindList 从指定绑定数组移除一条；返回是否命中
func (rm *Roadmap) UnbindList(r PointRef, list func(*Point) *[]string, rel string) bool {
	l := list(rm.PointAt(r))
	for i, s := range *l {
		if s == rel {
			*l = append((*l)[:i], (*l)[i+1:]...)
			return true
		}
	}
	return false
}

// MoveBinding 把 src 清单中的一条移挂到 dst 同类清单（卸载+挂载）；src 未命中返回 false
func (rm *Roadmap) MoveBinding(src PointRef, list func(*Point) *[]string, rel string, dst PointRef) bool {
	if !rm.UnbindList(src, list, rel) {
		return false
	}
	bindList(list(rm.PointAt(dst)), rel)
	return true
}

// IsBound 路径是否被任一知识点绑定（spring/code/river/ship/sea 任一）
func (rm *Roadmap) IsBound(rel string) bool {
	_, ok := rm.OwnerOf(rel)
	return ok
}

// OwnerOf 反查绑定路径所属知识点（树序首个命中）
func (rm *Roadmap) OwnerOf(rel string) (PointRef, bool) {
	for bi := range rm.Branches {
		for pi := range rm.Branches[bi].Points {
			p := &rm.Branches[bi].Points[pi]
			if p.Spring == rel || p.Code == rel {
				return PointRef{bi, pi}, true
			}
			for _, list := range [][]string{p.Sea, p.Ship, p.River} {
				for _, s := range list {
					if s == rel {
						return PointRef{bi, pi}, true
					}
				}
			}
		}
	}
	return PointRef{}, false
}
```

```go
// SpringPath spring 文件名约定路径（知识点标题全局唯一保证不冲突）
func SpringPath(title string) string { return "Study/spring/" + title + ".md" }

// CheckTitleUnique 新标题全树查重；空标题报错（供改名/拆分流程先验证再逐步应用）
func (rm *Roadmap) CheckTitleUnique(title string) error {
	title = strings.TrimSpace(title)
	if title == "" {
		return errors.New("标题不能为空")
	}
	for bi := range rm.Branches {
		for pi := range rm.Branches[bi].Points {
			if rm.Branches[bi].Points[pi].Title == title {
				return fmt.Errorf("知识点 %q 已存在于 %s——v2 要求全树唯一", title, rm.Branches[bi].Title)
			}
		}
	}
	return nil
}

// RenamePoint 知识点改名：查重后改标题；spring 字段符合约定路径时跟随改为新约定路径
// （文件改名由壳层负责）。返回改前 spring 路径：非空表示发生了约定跟随。
func (rm *Roadmap) RenamePoint(r PointRef, newTitle string) (string, error) {
	newTitle = strings.TrimSpace(newTitle)
	p := rm.PointAt(r)
	if newTitle == "" {
		return "", errors.New("标题不能为空")
	}
	if newTitle == p.Title {
		return "", nil
	}
	if err := rm.CheckTitleUnique(newTitle); err != nil {
		return "", err
	}
	old := p.Title
	p.Title = newTitle
	if p.Spring == SpringPath(old) {
		oldSpring := p.Spring
		p.Spring = SpringPath(newTitle)
		return oldSpring, nil
	}
	return "", nil
}

// maxStatus 状态进度取高：todo < doing < done
func maxStatus(a, b string) string {
	rank := map[string]int{StatusTodo: 0, StatusDoing: 1, StatusDone: 2}
	if rank[b] > rank[a] {
		return b
	}
	return a
}

// appendUnion 追加去重
func appendUnion(dst, src []string) []string {
	for _, s := range src {
		found := false
		for _, d := range dst {
			if d == s {
				found = true
				break
			}
		}
		if !found {
			dst = append(dst, s)
		}
	}
	return dst
}

// MergePoints 把 absorb 并入 keep 后删除 absorb：数组绑定合并去重；spring/code 空则补
//（双方都有时由壳层先裁决 keep 侧取值）；状态取进度高者；tags 并集；note 拼接。
// 调用后传入的 PointRef 下标失效，需按标题重新定位。
func (rm *Roadmap) MergePoints(keep, absorb PointRef) {
	kp, ap := rm.PointAt(keep), rm.PointAt(absorb)
	for _, rel := range ap.Sea {
		bindList(&kp.Sea, rel)
	}
	for _, rel := range ap.Ship {
		bindList(&kp.Ship, rel)
	}
	for _, rel := range ap.River {
		bindList(&kp.River, rel)
	}
	if kp.Spring == "" {
		kp.Spring = ap.Spring
	}
	if kp.Code == "" {
		kp.Code = ap.Code
	}
	kp.Status = maxStatus(kp.Status, ap.Status)
	kp.Tags = appendUnion(kp.Tags, ap.Tags)
	if ap.Note != "" {
		if kp.Note != "" {
			kp.Note += "\n---\n"
		}
		kp.Note += ap.Note
	}
	b := &rm.Branches[absorb.Branch]
	b.Points = append(b.Points[:absorb.Point], b.Points[absorb.Point+1:]...)
}

// SplitPoint 拆分：在 r 正后方插入同分组新知识点（todo、无绑定）；标题全树查重
func (rm *Roadmap) SplitPoint(r PointRef, newTitle string) (PointRef, error) {
	if err := rm.CheckTitleUnique(newTitle); err != nil {
		return PointRef{}, err
	}
	src := rm.PointAt(r)
	np := Point{Title: strings.TrimSpace(newTitle), Group: src.Group, Status: StatusTodo}
	b := &rm.Branches[r.Branch]
	b.Points = append(b.Points[:r.Point+1], append([]Point{np}, b.Points[r.Point+1:]...)...)
	return PointRef{r.Branch, r.Point + 1}, nil
}

// MovePointToBranch 换主干：挪到目标主干末尾；同主干为空操作
func (rm *Roadmap) MovePointToBranch(r PointRef, bi int) {
	if r.Branch == bi {
		return
	}
	pt := *rm.PointAt(r) // 先值拷贝：删除会使源切片内存移位
	rm.Branches[r.Branch].Points = append(rm.Branches[r.Branch].Points[:r.Point], rm.Branches[r.Branch].Points[r.Point+1:]...)
	rm.Branches[bi].Points = append(rm.Branches[bi].Points, pt)
}

// SetGroup 换分组标签（空串清除）
func (rm *Roadmap) SetGroup(r PointRef, group string) { rm.PointAt(r).Group = group }

// PointEmpty 知识点五类绑定全空
func (rm *Roadmap) PointEmpty(r PointRef) bool {
	p := rm.PointAt(r)
	return p.Spring == "" && p.Code == "" && len(p.Sea) == 0 && len(p.Ship) == 0 && len(p.River) == 0
}

// DeletePoint 删除知识点；仍有任一绑定时拒绝（壳层先提示清理）
func (rm *Roadmap) DeletePoint(r PointRef) error {
	if !rm.PointEmpty(r) {
		return errors.New("知识点仍有绑定，先卸载后再删除")
	}
	b := &rm.Branches[r.Branch]
	b.Points = append(b.Points[:r.Point], b.Points[r.Point+1:]...)
	return nil
}
```

```go
var structureMenu = []string{"合并", "拆分", "换主干", "换分组", "删除"}

// listAcc 数组型三类绑定访问器
func listAcc(t Tier) func(*roadmap.Point) *[]string {
	switch t {
	case TierSea:
		return roadmap.SeaList
	case TierShip:
		return roadmap.ShipList
	case TierRiver:
		return roadmap.RiverList
	}
	return nil

// bindSingle 把 rel 绑定挂到 ref 的对应清单
func bindSingle(m *Model, ref roadmap.PointRef, t Tier, rel string) {
	switch t {
	case TierSpring:
		m.RM.BindSpring(ref, rel)
	case TierCode:
		m.RM.BindCode(ref, rel)
	case TierSea:
		m.RM.BindSea(ref, rel)
	case TierShip:
		m.RM.BindShip(ref, rel)
	case TierRiver:
		m.RM.BindRiver(ref, rel)
	}
}

// bindMove 五类通用移动；spring/code 目标已占用返回 false
func bindMove(m *Model, src roadmap.PointRef, t Tier, rel string, dst roadmap.PointRef) bool {
	switch t {
	case TierSpring:
		if m.RM.PointAt(dst).Spring != "" {
			return false
		}
		m.RM.UnbindSpring(src)
		m.RM.BindSpring(dst, rel)
	case TierCode:
		if m.RM.PointAt(dst).Code != "" {
			return false
		}
		m.RM.UnbindCode(src)
		m.RM.BindCode(dst, rel)
	default:
		return m.RM.MoveBinding(src, listAcc(t), rel, dst)
	}
	return true
}

// save 落盘并刷新待归类缓存；失败写错误行不中断
func (m *Model) save(h Hooks) {
	if h.Save != nil {
		if err := h.Save(m.RM); err != nil {
			m.Err = "落盘失败：" + err.Error()
		}
	}
	m.refreshUnbound()
}

// locate 结构操作后按标题重新定位（标题全树唯一）；找不到不动
func (m *Model) locate(title string) {
	for bi := range m.RM.Branches {
		for pi := range m.RM.Branches[bi].Points {
			if m.RM.Branches[bi].Points[pi].Title == title {
				m.Branch, m.Point = bi, pi
				m.Expanded[bi] = true
				m.Normalize()
				return
			}
		}
	}
}

// doUnbind 卸载光标绑定（详情页笔记行）；只动记录，文件留磁盘
func doUnbind(m *Model, h Hooks) {
	if m.View != viewPoint {
		m.Err = "详情页才能卸载绑定"
		return
	}
	r := m.rowAt(m.Cursor)
	if r == nil || r.Kind != rowNote {
		m.Err = "光标不在绑定条目上"
		return
	}
	ref := m.curRef()
	switch r.Tier {
	case TierSpring:
		m.RM.UnbindSpring(ref)
	case TierCode:
		m.RM.UnbindCode(ref)
	default:
		if !m.RM.UnbindList(ref, listAcc(r.Tier), r.Note) {
			m.Err = "绑定不存在"
			return
		}
	}
	m.save(h)
}

// doMove 移动光标绑定到其他知识点（详情页笔记行；可跨主干）
func doMove(m *Model, h Hooks) {
	r := m.rowAt(m.Cursor)
	if m.View != viewPoint || r == nil || r.Kind != rowNote {
		m.Err = "光标不在绑定条目上"
		return
	}
	if h.SelectPoint == nil {
		m.Err = "该操作不可用"
		return
	}
	dst, ok := h.SelectPoint("移动到哪个知识点？")
	if !ok {
		return
	}
	src := m.curRef()
	if src == dst {
		m.Err = "目标就是当前知识点"
		return
	}
	if r.Tier == TierSpring || r.Tier == TierCode {
		if !bindMove(m, src, r.Tier, r.Note, dst) {
			m.Err = "目标知识点已有该项，先卸载后再移动"
			return
		}
	} else if !m.RM.MoveBinding(src, listAcc(r.Tier), r.Note, dst) {
		m.Err = "绑定不存在"
		return
	}
	m.save(h)
}

// doAttach 挂载未归类文件：详情页=挂到光标清单节；待归类区=挂到选定知识点
func doAttach(m *Model, h Hooks) {
	switch m.View {
	case viewPoint:
		tier, ok := m.TierAtCursor()
		if !ok {
			m.Err = "光标不在清单节内"
			return
		}
		ref := m.curRef()
		if (tier == TierSpring && m.RM.PointAt(ref).Spring != "") ||
			(tier == TierCode && m.RM.PointAt(ref).Code != "") {
			m.Err = "本知识点已有该项，先卸载再挂载"
			return
		}
		m.refreshUnbound()
		rel, ok := pickUnbound(m, h, tier)
		if !ok {
			return
		}
		bindSingle(m, ref, tier, rel)
		m.save(h)
	case viewUnbound:
		r := m.rowAt(m.Cursor)
		if r == nil || r.Kind != rowNote {
			m.Err = "光标不在文件行上"
			return
		}
		if h.SelectPoint == nil {
			m.Err = "该操作不可用"
			return
		}
		dst, ok := h.SelectPoint("挂载到哪个知识点？")
		if !ok {
			return
		}
		if m.RM.IsBound(r.Note) { // 快照后已被他处挂载：转为移动
			owner, _ := m.RM.OwnerOf(r.Note)
			if !h.Confirm("该文件已挂载于「" + m.RM.PathOf(owner) + "」，改为移动？") {
				return
			}
			if !bindMove(m, owner, r.Tier, r.Note, dst) {
				m.Err = "目标知识点已有该项，操作取消"
				return
			}
		} else {
			if (r.Tier == TierSpring && m.RM.PointAt(dst).Spring != "") ||
				(r.Tier == TierCode && m.RM.PointAt(dst).Code != "") {
				m.Err = "目标知识点已有该项，先卸载再挂载"
				return
			}
			bindSingle(m, dst, r.Tier, r.Note)
		}
		m.refreshUnbound()
		m.save(h)
	default:
		m.Err = "该视图不支持挂载"
	}
}

// pickUnbound 从未归类清单挑一个文件；空清单报错不弹菜单

}

// doSplit 拆分：新点插正后方（同组、todo）；sea/ship/river 逐清单挑选移入；
// 选择阶段不改 RM，放弃（Esc）即整体取消，不产生半截变更
func doSplit(m *Model, h Hooks) {
	src, ok := m.PointAtCursor()
	if !ok {
		m.Err = "光标不在知识点上"
		return
	}
	if h.Input == nil || h.SelectOne == nil {
		m.Err = "该操作不可用"
		return
	}
	title, ok := h.Input("新知识点标题（拆自「" + m.RM.PointAt(src).Title + "」）")
	if !ok {
		return
	}
	if err := m.RM.CheckTitleUnique(title); err != nil {
		m.Err = err.Error()
		return
	}
	type allocItem struct {
		t   Tier
		rel string
	}
	var allocs []allocItem
	picked := map[string]bool{}
	for _, t := range []Tier{TierSea, TierShip, TierRiver} {
		for {
			var remain []string
			for _, rel := range listAcc(t)(m.RM.PointAt(src)) {
				if !picked[rel] {
					remain = append(remain, rel)
				}
			}
			if len(remain) == 0 {
				break
			}
			opts := append(append([]string{}, remain...), "（完成本清单）")
			idx, ok := h.SelectOne("「"+t.Label()+"」哪些移到新知识点？（Esc 取消整个拆分）", opts)
			if !ok {
				return
			}
			if idx == len(remain) {
				break
			}
			picked[remain[idx]] = true
			allocs = append(allocs, allocItem{t, remain[idx]})
		}
	}
	srcTitle := m.RM.PointAt(src).Title
	np, err := m.RM.SplitPoint(src, title)
	if err != nil {
		m.Err = err.Error()
		return
	}
	for _, a := range allocs {
		m.RM.MoveBinding(src, listAcc(a.t), a.rel, np)
	}
	m.locate(srcTitle)
	m.save(h)
}

}
```

```go
func pauseAndAct(out io.Writer, sess *keySession, fn func()) error {
	sess.close()

	}
	return nil
```

```go
// tierRoot 待归类扫描的 Study 子目录
var tierRoot = map[planview.Tier]string{
	planview.TierSea:    "sea",
	planview.TierShip:   "ship",
	planview.TierRiver:  "river",
	planview.TierSpring: "spring",
	planview.TierCode:   "code",
}

// scanUnbound 扫描 Study/<tier> 下未被任何知识点绑定的文件（code 认 .go，其余 .md），按路径排序；
// 目录缺失按 0 计（不算错误）
func scanUnbound(root string, rm *roadmap.Roadmap, t planview.Tier) []string {
	dir := filepath.Join(root, "Study", tierRoot[t])
	ext := ".md"
	if t == planview.TierCode {
		ext = ".go"
	}
	var out []string
	err := filepath.WalkDir(dir, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // 目录不存在/无权限：跳过
		}
		if d.IsDir() || !strings.EqualFold(filepath.Ext(d.Name()), ext) {
			return nil
		}
		rel, rerr := filepath.Rel(root, path)
		if rerr != nil {
			return nil
		}
		rel = filepath.ToSlash(rel)
		if !rm.IsBound(rel) {
			out = append(out, rel)
		}
		return nil
	})
	if err != nil {
		return nil
	}
	return out
}

// selectPointFlow 关键词过滤选知识点：输入关键词（空=全部）→ 唯一直达/多选/零匹配重试
func selectPointFlow(ctx *Ctx, rm *roadmap.Roadmap, prompt string) (roadmap.PointRef, bool) {
	for {
		kw, err := inputLine(ctx, prompt+"（关键词过滤，空=全部，Esc 取消）")
		if err != nil {
			return roadmap.PointRef{}, false
		}
		hits := rm.MatchPoints(strings.TrimSpace(kw))
		switch len(hits) {
		case 0:
			fmt.Fprintln(ctx.Out, "没有匹配的知识点，换个关键词或 Esc 取消")
			continue
		case 1:
			return hits[0], true
		}
		opts := make([]menuOption, 0, len(hits))
		for _, h := range hits {
			opts = append(opts, option(rm.PathOf(h), strconv.Itoa(h.Branch)+":"+strconv.Itoa(h.Point)))
		}
		var choice string
		if err := selectFrom(ctx, "选择知识点", opts, &choice); err != nil {
			return roadmap.PointRef{}, false
		}
		var ref roadmap.PointRef
		fmt.Sscanf(choice, "%d:%d", &ref.Branch, &ref.Point)
		return ref, true
	}
}

// selectOneFlow 单选菜单，返回下标
func selectOneFlow(ctx *Ctx, prompt string, opts []string) (int, bool) {
	mo := make([]menuOption, 0, len(opts))
	for i, o := range opts {
		mo = append(mo, option(o, strconv.Itoa(i)))
	}
	var choice string
	if err := selectFrom(ctx, prompt, mo, &choice); err != nil {
		return 0, false
	}
	i, err := strconv.Atoi(choice)
	if err != nil {
		return 0, false
	}
	return i, true
}
```

```go
			_ = openPlanNote(ctx, rel)
		},
		Save: func(rm *roadmap.Roadmap) error { return roadmap.Save(path, rm) },
		SelectPoint: func(prompt string) (roadmap.PointRef, bool) {
			return selectPointFlow(ctx, rm, prompt)
		},
		Input: func(prompt string) (string, bool) {
			// 空串透传：换分组「留空清除」依赖空串；改名空标题由 RenamePoint 校验兜底
			s, err := inputLine(ctx, prompt)
			return s, err == nil
		},
		Confirm: func(summary string) bool {
			ok, err := confirm(ctx, summary)
			return err == nil && ok
		},
		SelectOne: func(prompt string, opts []string) (int, bool) {
			return selectOneFlow(ctx, prompt, opts)
		},
		Unbound: func(t planview.Tier) []string { return scanUnbound(ctx.WS.Root, rm, t) },
		RenameFile: func(oldRel, newRel string) error {
			return os.Rename(
				filepath.Join(ctx.WS.Root, filepath.FromSlash(oldRel)),
				filepath.Join(ctx.WS.Root, filepath.FromSlash(newRel)))
		},
	})
```
