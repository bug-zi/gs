**Architecture:** `internal/roadmap` v2 重构（`branch → point`，全局唯一；`spring` 单值 + `river`/`sea` 数组）；新建 `internal/planview`（纯状态机 + 整帧 ANSI 渲染 + 浏览循环），raw 单键复用 `internal/termkey.KeyReader`（并行会话已提供），交互模式照抄 `project_tree_view.go` 范本（keySession 暂停/恢复、视口滑动、反白光标、`\r\x1b[J` 整帧重绘、非 TTY 降级）；`command/plan.go` 只做装配与回调。

**实现期决策（本计划锁定）：**

1. **roadmap v2**：`Version=2`；v1 文件解析报错并提示「v1 已废弃，用 v2 初稿覆盖」；`topic` 层删除，知识点 `title` **全局唯一**（Validate 查重）。
2. **绑定字段**：`spring`（单值 string）/`river`/`sea`（数组）一律工作区相对路径；Bind 系列幂等；失效显示 `[失效]` 不自动清理。
3. **静态渲染保留为降级通道**：`Render/RenderOpts` 签名不变，输出加分组小标题行；`/plan <主干>` 非 TTY 时走它。
4. **planview 三视图**：全局树（主干行 Tab 展开/收起、Enter 进主干页；知识点行 Enter 直进详情）/ 主干页（分组小标题 + 知识点，Enter 进详情）/ 详情页（spring×0/1、river×N、sea×N 清单，Enter/o 打开笔记）。Esc 逐层返回（全局树 Esc=退出）；q/Ctrl-C/Ctrl-D 任意层退出。
5. **d/s/t**：三个视图里凡光标语义命中知识点（分支页/详情页恒定命中当前知识点；全局树命中光标知识点行）即改状态，经 `Hooks.Save` 回调原子落盘，失败显示底部错误行不退出。
6. **raw 暂停/恢复**：打开笔记（pager 为行式输入）前 `KeyReader.Close()`、返回后重开；失败降级静态输出——照抄 `project_tree_view.go` 的 keySession 模式。
7. **`/spring add` 重构**：`/spring add <知识点关键词>` 定位知识点 → `Study/spring/<知识点标题>.md` 已存在则直接打开、否则创建占位条目并 `BindSpring` 落盘 → 打开编辑器。它是 spring 的唯一创建入口。
8. **`/river add` 尾部追问**：创建 river 条目后问「挂到知识点？（关键词，回车跳过）」，命中即 `BindRiver` 落盘；追问的匹配逻辑抽纯函数可测。
9. **`/plan done/doing/todo <关键词>`** 保留为参数快捷通道（`MatchPoints` 三态）；无参报错提示用浏览器 d/s/t。`/plan add` 参数改 `branch|point`。
10. **测试边界**：planview 状态机/按键动作映射/帧渲染（stripANSI）表驱动可测；raw 读循环与 huh 交互壳不自动测（惯例），走真机冒烟。

// Package roadmap Go 学习路线图的纯逻辑：TOML 正本解析/写回、关键词匹配、状态与绑定变更、静态树渲染。
// 本包不 import ui/lipgloss，颜色由调用方注入。
package roadmap

import (
	"bytes"
	"errors"
	"fmt"
	"os"

	"github.com/BurntSushi/toml"

	"github.com/bugzi/gs/internal/workspace"
)

const (
	// Version 路线图文件格式版本（v2：branch→point，topic 层废止）
	Version = 2

	StatusTodo  = "todo"
	StatusDoing = "doing"
	StatusDone  = "done"
)

// Point 知识点（二级）；绑定路径一律工作区相对路径，Spring 为 0/1 深度绑定
type Point struct {
	Title  string   `toml:"title"`
	Group  string   `toml:"group,omitempty"`
	Status string   `toml:"status"` // todo(缺省) | doing | done
	Tags   []string `toml:"tags,omitempty"`
	Spring string   `toml:"spring,omitempty"`
	River  []string `toml:"river,omitempty"`
	Sea    []string `toml:"sea,omitempty"`
	Note   string   `toml:"note,omitempty"`
}

// Branch 主干（一级）
type Branch struct {
	Title  string  `toml:"title"`
	Points []Point `toml:"point,omitempty"`
}

// Roadmap 树状路线图正本
type Roadmap struct {
	Version  int      `toml:"version"`
	Branches []Branch `toml:"branch"`
}

// Load 读入并校验 roadmap.toml
func Load(path string) (*Roadmap, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	return Parse(data)
}

// Parse 解析；TOML 语法错误带行列号返回
func Parse(data []byte) (*Roadmap, error) {
	var rm Roadmap
	if err := toml.Unmarshal(data, &rm); err != nil {
		var perr toml.ParseError
		if errors.As(err, &perr) {
			return nil, fmt.Errorf("roadmap.toml 解析失败：\n%s", perr.ErrorWithPosition())
		}
		return nil, err
	}
	if err := rm.Validate(); err != nil {
		return nil, err
	}
	return &rm, nil
}

// Validate 结构校验：版本、主干非空、标题非空、知识点全局唯一、status 合法（缺省归一化）
func (rm *Roadmap) Validate() error {
	switch rm.Version {
	case Version:
	case 1:
		return errors.New("检测到 v1 格式（branch.topic.node）：roadmap.toml 已升级 v2（branch.point），请用 v2 初稿覆盖（初稿见 docs/project/plan/2026-10-26-路线图v2内容清单.md）")
	default:
		return fmt.Errorf("不支持的版本 %d（期望 %d）", rm.Version, Version)
	}
	if len(rm.Branches) == 0 {
		return errors.New("路线图没有任何主干")
	}
	seen := map[string]string{} // 知识点标题 → 主干，全局唯一
	for bi := range rm.Branches {
		b := &rm.Branches[bi]
		if b.Title == "" {
			return fmt.Errorf("第 %d 个主干缺少标题", bi+1)
		}
		for pi := range b.Points {
			p := &b.Points[pi]
			if p.Title == "" {
				return fmt.Errorf("主干 %q 存在缺少标题的知识点", b.Title)
			}
			if prev, dup := seen[p.Title]; dup {
				return fmt.Errorf("知识点 %q 重复（%s 与 %s）——v2 要求全树唯一", p.Title, prev, b.Title)
			}
			seen[p.Title] = b.Title
			switch p.Status {
			case "":
				p.Status = StatusTodo
			case StatusTodo, StatusDoing, StatusDone:
			default:
				return fmt.Errorf("知识点 %q 状态非法 %q（todo/doing/done）", p.Title, p.Status)
			}
		}
	}
	return nil
}

// Save 原子写回（重编码会丢弃手写注释，命令维护为主属预期）
func Save(path string, rm *Roadmap) error {
	var buf bytes.Buffer
	if err := toml.NewEncoder(&buf).Encode(rm); err != nil {
		return err
	}
	return workspace.WriteFileAtomic(path, buf.Bytes())
}

package roadmap

import (
	"errors"
	"fmt"
	"strings"
)

// PointRef 知识点定位：主干/知识点两段下标
type PointRef struct{ Branch, Point int }

// PointAt 取知识点指针
func (rm *Roadmap) PointAt(r PointRef) *Point {
	return &rm.Branches[r.Branch].Points[r.Point]
}

// PathOf 知识点全路径「主干 › 知识点」
func (rm *Roadmap) PathOf(r PointRef) string {
	return rm.Branches[r.Branch].Title + " › " + rm.PointAt(r).Title
}

// MatchPoints 关键词匹配全树知识点标题（大小写不敏感的包含匹配），按树序返回
func (rm *Roadmap) MatchPoints(kw string) []PointRef {
	kw = strings.ToLower(kw)
	var out []PointRef
	for bi := range rm.Branches {
		for pi := range rm.Branches[bi].Points {
			if strings.Contains(strings.ToLower(rm.Branches[bi].Points[pi].Title), kw) {
				out = append(out, PointRef{bi, pi})
			}
		}
	}
	return out
}

// MatchBranches 主干标题含关键词的下标
func (rm *Roadmap) MatchBranches(kw string) []int {
	kw = strings.ToLower(kw)
	var out []int
	for i := range rm.Branches {
		if strings.Contains(strings.ToLower(rm.Branches[i].Title), kw) {
			out = append(out, i)
		}
	}
	return out
}

// SetStatus 标记知识点状态
func (rm *Roadmap) SetStatus(r PointRef, status string) error {
	switch status {
	case StatusTodo, StatusDoing, StatusDone:
	default:
		return fmt.Errorf("非法状态 %q", status)
	}
	rm.PointAt(r).Status = status
	return nil
}

// TotalPoints / DonePoints 全树知识点数与 done 数
func (rm *Roadmap) TotalPoints() int {
	n := 0
	for bi := range rm.Branches {
		n += len(rm.Branches[bi].Points)
	}
	return n
}

func (rm *Roadmap) DonePoints() int {
	n := 0
	for bi := range rm.Branches {
		for pi := range rm.Branches[bi].Points {
			if rm.Branches[bi].Points[pi].Status == StatusDone {
				n++
			}
		}
	}
	return n
}

// BranchProgress 主干内 done/total
func (rm *Roadmap) BranchProgress(bi int) (done, total int) {
	for pi := range rm.Branches[bi].Points {
		total++
		if rm.Branches[bi].Points[pi].Status == StatusDone {
			done++
		}
	}
	return done, total
}

// AddBranch 新主干，重名报错
func (rm *Roadmap) AddBranch(title string) error {
	if strings.TrimSpace(title) == "" {
		return errors.New("标题不能为空")
	}
	for i := range rm.Branches {
		if rm.Branches[i].Title == title {
			return fmt.Errorf("主干已存在: %s", title)
		}
	}
	rm.Branches = append(rm.Branches, Branch{Title: title})
	return nil
}

// AddPoint 主干下新知识点（status=todo）；标题全树唯一（spring 文件名约定依赖此约束）
func (rm *Roadmap) AddPoint(bi int, title, group string, tags []string) error {
	if strings.TrimSpace(title) == "" {
		return errors.New("标题不能为空")
	}
	for bi2 := range rm.Branches {
		for pi := range rm.Branches[bi2].Points {
			if rm.Branches[bi2].Points[pi].Title == title {
				return fmt.Errorf("知识点 %q 已存在于 %s——v2 要求全树唯一", title, rm.Branches[bi2].Title)
			}
		}
	}
	rm.Branches[bi].Points = append(rm.Branches[bi].Points, Point{Title: title, Group: group, Status: StatusTodo, Tags: tags})
	return nil
}

// BindSpring 绑定唯一 spring（覆盖语义，0/1）
func (rm *Roadmap) BindSpring(r PointRef, rel string) {
	rm.PointAt(r).Spring = rel
}

// BindRiver / BindSea 追加绑定（幂等）
func (rm *Roadmap) BindRiver(r PointRef, rel string) { bindList(&rm.PointAt(r).River, rel) }
func (rm *Roadmap) BindSea(r PointRef, rel string)   { bindList(&rm.PointAt(r).Sea, rel) }

func bindList(list *[]string, rel string) {
	for _, s := range *list {
		if s == rel {
			return
		}
	}
	*list = append(*list, rel)
}

// ColorFn 状态行上色回调：返回包装后的行；nil 表示不上色。
// 由壳层注入（lipgloss），本包不 import ui，保持纯逻辑可独立测试。
type ColorFn func(status, line string) string

// RenderOpts 静态渲染选项（浏览器不可用时的降级通道）
type RenderOpts struct {
	Title     string  // 树标题（如「Go 学习路线图」）
	Progress  string  // 树顶进度段，空则不显示
	Next      string  // 树顶断点摘要行，空则不显示
	BranchIdx []int   // 只渲染这些主干；nil = 全部
	Color     ColorFn // nil = 不上色
}

// Glyph 状态符号：done [x] / doing [~] / 其他 [ ]
func Glyph(status string) string {
	switch status {
	case StatusDone:
		return "[x]"
	case StatusDoing:
		return "[~]"
	default:
		return "[ ]"
	}
}

// Render 静态树：主干行挂 done/total（空主干不挂）；分组标签变化时输出小标题行；
// 「高频」尾随 *；主干过滤只输出指定主干。
func Render(rm *Roadmap, o RenderOpts) string {
	var b strings.Builder
	if o.Progress != "" {
		b.WriteString(leaderLine(o.Title, o.Progress, 44) + "\n")
	} else {
		b.WriteString(o.Title + "\n")
	}
	if o.Next != "" {
		b.WriteString(o.Next + "\n")
	}
	b.WriteString(strings.Repeat("═", 44) + "\n")

	ids := o.BranchIdx
	if ids == nil {
		ids = make([]int, len(rm.Branches))
		for i := range rm.Branches {
			ids[i] = i
		}
	}
	for k, bi := range ids {
		branch := &rm.Branches[bi]
		bg, bc := "├─ ", "│  "
		if k == len(ids)-1 {
			bg, bc = "└─ ", "   "
		}
		line := bg + branch.Title
		if done, total := rm.BranchProgress(bi); total > 0 {
			line = leaderLine(line, fmt.Sprintf("%d/%d", done, total), 40)
		}
		b.WriteString(line + "\n")
		prevGroup := ""
		for pi := range branch.Points {
			p := &branch.Points[pi]
			if p.Group != "" && p.Group != prevGroup {
				b.WriteString(bc + "▸ " + p.Group + "\n")
				prevGroup = p.Group
			} else if p.Group == "" {
				prevGroup = ""
			}
			ng := "├─ "
			if pi == len(branch.Points)-1 {
				ng = "└─ "
			}
			nl := bc + "   " + ng + Glyph(p.Status) + " " + p.Title
			if hasHighFreq(p.Tags) {
				nl += " *"
			}
			if o.Color != nil {
				nl = o.Color(p.Status, nl)
			}
			b.WriteString(nl + "\n")
		}
	}
	return b.String()
}

// leaderLine 「title ···· progress」三段行，目标总显示宽 w（中点至少 3 个）
func leaderLine(title, progress string, w int) string {
	dots := w - runewidth.StringWidth(title) - runewidth.StringWidth(progress) - 2
	if dots < 3 {
		dots = 3
	}
	return title + " " + strings.Repeat("·", dots) + " " + progress
}

// hasHighFreq tags 含「高频」即视为高频
func hasHighFreq(tags []string) bool {
	for _, tg := range tags {
		if strings.Contains(tg, "高频") {
			return true
		}
	}
	return false
}

package planview

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/bugzi/gs/internal/roadmap"
)

type viewKind int

const (
	viewGlobal viewKind = iota // 全局树：主干 + 展开的知识点
	viewBranch                 // 主干页：单主干知识点列表
	viewPoint                  // 知识点详情：spring/river/sea 清单
)

type rowKind int

const (
	rowBranchHead rowKind = iota // 主干行（可选）
	rowGroupHead                 // 分组小标题（不可选）
	rowPoint                     // 知识点行（可选）
	rowSectionHead               // 详情页小节标题（不可选）
	rowNote                      // 笔记路径行（可选，可打开）
	rowHint                      // 提示行（不可选）
)

type row struct {
	Kind       rowKind
	Branch     int
	Point      int
	Note       string // rowNote：绑定路径
	Selectable bool
	Text       string
}

// Model 浏览器状态；RM 原地变更，落盘经 Hooks.Save 由壳层回调
type Model struct {
	RM       *roadmap.Roadmap
	Root     string // 工作区根（失效判断用）
	Exists   func(rel string) bool // nil = os.Stat 默认实现（测试注入）
	View     viewKind
	Branch   int
	Point    int
	Cursor   int
	Expanded map[int]bool
	Err      string // 底部错误行（落盘失败等），按键后清除
}

// Rows 依当前视图计算可见行
func (m *Model) Rows() []row {
	switch m.View {
	case viewGlobal:
		return m.globalRows()
	case viewBranch:
		return m.branchRows()
	default:
		return m.pointRows()
	}
}

func (m *Model) globalRows() []row {
	var rows []row
	for bi := range m.RM.Branches {
		b := &m.RM.Branches[bi]
		mark := "▸ "
		if m.Expanded[bi] {
			mark = "▾ "
		}
		text := mark + b.Title
		if done, total := m.progress(bi); total > 0 {
			text = leader(text, fmt.Sprintf("%d/%d", done, total))
		}
		rows = append(rows, row{Kind: rowBranchHead, Branch: bi, Selectable: true, Text: text})
		if !open(bi, m.Expanded) {
			continue
		}
		rows = appendPointRows(rows, b, bi, "    ")
	}
	return rows
}

func (m *Model) branchRows() []row {
	b := &m.RM.Branches[m.Branch]
	return appendPointRows(nil, b, m.Branch, "  ")
}

// appendPointRows 追加知识点行；分组标签变化时插入小标题行（同组连续只出一次）
func appendPointRows(rows []row, b *roadmap.Branch, bi int, indent string) []row {
	prev := ""
	for pi := range b.Points {
		p := &b.Points[pi]
		if p.Group != "" && p.Group != prev {
			rows = append(rows, row{Kind: rowGroupHead, Branch: bi, Text: indent + p.Group})
			prev = p.Group
		} else if p.Group == "" {
			prev = ""
		}
		rows = append(rows, row{Kind: rowPoint, Branch: bi, Point: pi, Selectable: true,
			Text: indent + "  " + roadmap.Glyph(p.Status) + " " + p.Title + star(p.Tags)})
	}
	return rows
}

func (m *Model) pointRows() []row {
	p := m.curPoint()
	var rows []row
	rows = append(rows, row{Kind: rowSectionHead, Text: "spring · 我的总结 ×0/1"})
	if p.Spring == "" {
		rows = append(rows, row{Kind: rowHint, Text: "  （尚未创建，/spring add " + p.Title + " 可创建并绑定）"})
	} else {
		rows = append(rows, m.noteRow(p.Spring))
	}
	rows = append(rows, row{Kind: rowSectionHead, Text: fmt.Sprintf("river · 摘录 ×%d", len(p.River))})
	for _, rel := range p.River {
		rows = append(rows, m.noteRow(rel))
	}
	if len(p.River) == 0 {
		rows = append(rows, row{Kind: rowHint, Text: "  （无；/river add 摘录后可挂载）"})
	}
	rows = append(rows, row{Kind: rowSectionHead, Text: fmt.Sprintf("sea · 来源 ×%d", len(p.Sea))})
	for _, rel := range p.Sea {
		rows = append(rows, m.noteRow(rel))
	}
	if len(p.Sea) == 0 {
		rows = append(rows, row{Kind: rowHint, Text: "  （无）"})
	}
	return rows
}

func (m *Model) noteRow(rel string) row {
	text := "  " + rel
	if !m.exists(rel) {
		text += "　[失效]"
	}
	return row{Kind: rowNote, Note: rel, Selectable: true, Text: text}
}

func (m *Model) exists(rel string) bool {
	if m.Exists != nil {
		return m.Exists(rel)
	}
	_, err := os.Stat(filepath.Join(m.Root, filepath.FromSlash(rel)))
	return err == nil
}

// Normalize 光标夹取与可选行校正：越界拉回、落在不可选行时移到最近可选行
func (m *Model) Normalize() {
	rows := m.Rows()
	if len(rows) == 0 {
		m.Cursor = 0
		return
	}
	if m.Cursor >= len(rows) {
		m.Cursor = len(rows) - 1
	}
	if m.Cursor < 0 {
		m.Cursor = 0
	}
	if rows[m.Cursor].Selectable {
		return
	}
	for i := m.Cursor; i < len(rows); i++ {
		if rows[i].Selectable {
			m.Cursor = i
			return
		}
	}
	for i := m.Cursor; i >= 0; i-- {
		if rows[i].Selectable {
			m.Cursor = i
			return
		}
	}
}

// Move 光标移动；移到头即停
func (m *Model) Move(delta int) { m.Cursor += delta; m.Normalize() }

// Toggle 全局树主干行展开/收起；其余情况无效
func (m *Model) Toggle() {
	if m.View != viewGlobal {
		return
	}
	if r := m.rowAt(m.Cursor); r != nil && r.Kind == rowBranchHead {
		m.Expanded[r.Branch] = !m.Expanded[r.Branch]
	}
}

// Action Enter 的语义结果
type Action int

const (
	ActNone Action = iota
	ActIntoBranch
	ActIntoPoint
	ActOpenNote
)

// Enter 进入：主干行→主干页；知识点行→详情页；笔记行→打开。返回动作与笔记路径
func (m *Model) Enter() (Action, string) {
	r := m.rowAt(m.Cursor)
	if r == nil || !r.Selectable {
		return ActNone, ""
	}
	switch {
	case m.View == viewGlobal && r.Kind == rowBranchHead:
		m.Branch, m.View, m.Cursor = r.Branch, viewBranch, 0
		m.Normalize()
		return ActIntoBranch, ""
	case m.View == viewGlobal && r.Kind == rowPoint:
		m.Branch, m.Point, m.View, m.Cursor = r.Branch, r.Point, viewPoint, 0
		m.Normalize()
		return ActIntoPoint, ""
	case m.View == viewBranch && r.Kind == rowPoint:
		m.Point, m.View, m.Cursor = r.Point, viewPoint, 0
		m.Normalize()
		return ActIntoPoint, ""
	case m.View == viewPoint && r.Kind == rowNote:
		return ActOpenNote, r.Note
	}
	return ActNone, ""
}

// Back 返回上一层；全局树返回 false（顶层，调用方退出浏览器）
func (m *Model) Back() bool {
	switch m.View {
	case viewPoint:
		m.View, m.Cursor = viewBranch, 0
		m.Normalize()
		return true
	case viewBranch:
		m.View, m.Cursor = viewGlobal, 0
		m.Normalize()
		return true
	}
	return false
}

// Mark 状态标记：详情页恒定命中当前知识点，其余视图命中光标知识点行。
// 返回命中的知识点与是否命中。
func (m *Model) Mark(status string) (roadmap.PointRef, bool) {
	if m.View == viewPoint {
		ref := roadmap.PointRef{m.Branch, m.Point}
		_ = m.RM.SetStatus(ref, status)
		return ref, true
	}
	if r := m.rowAt(m.Cursor); r != nil && r.Kind == rowPoint {
		ref := roadmap.PointRef{r.Branch, r.Point}
		_ = m.RM.SetStatus(ref, status)
		return ref, true
	}
	return roadmap.PointRef{}, false
}

// Focus 全局树中定位关键词主干：展开并移动光标到其主干行；找不到不动
func (m *Model) Focus(kw string) {
	kw = strings.ToLower(kw)
	for bi := range m.RM.Branches {
		if !strings.Contains(strings.ToLower(m.RM.Branches[bi].Title), kw) {
			continue
		}
		m.Expanded[bi] = true
		for i, r := range m.Rows() {
			if r.Kind == rowBranchHead && r.Branch == bi {
				m.Cursor = i
				return
			}
		}
		return
	}
}

func (m *Model) rowAt(i int) *row {
	rows := m.Rows()
	if i < 0 || i >= len(rows) {
		return nil
	}
	return &rows[i]
}

func (m *Model) curPoint() *roadmap.Point {
	return &m.RM.Branches[m.Branch].Points[m.Point]
}

func (m *Model) progress(bi int) (done, total int) {
	for pi := range m.RM.Branches[bi].Points {
		total++
		if m.RM.Branches[bi].Points[pi].Status == roadmap.StatusDone {
			done++
		}
	}
	return done, total
}

func open(bi int, expanded map[int]bool) bool { return expanded[bi] }

func star(tags []string) string {
	for _, tg := range tags {
		if strings.Contains(tg, "高频") {
			return " *"
		}
	}
	return ""
}

func leader(title, progress string) string {
	w := runewidth.StringWidth(title) + runewidth.StringWidth(progress)
	dots := 40 - w - 2
	if dots < 3 {
		dots = 3
	}
	return title + " " + strings.Repeat("·", dots) + " " + progress
}

import (
	"fmt"

// window 视口窗口：不超过 avail 全显；超出则以光标为中心滑动，贴边取齐
func window(n, cur, avail int) (lo, hi int) {
	if avail <= 0 || n <= avail {
		return 0, n
	}
	lo = cur - avail/2
	if lo < 0 {
		lo = 0
	}
	if lo+avail > n {
		lo = n - avail
	}
	return lo, lo + avail
}

// renderRow 单行渲染：超宽截断；光标行整行反白；小标题加粗；提示行灰
func renderRow(r row, selected bool, width int) string {
	if runewidth.StringWidth(r.Text) > width {
		r.Text = runewidth.Truncate(r.Text, width, "…")
	}
	switch {
	case selected:
		return "\x1b[7m" + r.Text + "\x1b[27m"
	case r.Kind == rowGroupHead || r.Kind == rowSectionHead:
		return "\x1b[1m" + r.Text + "\x1b[22m"
	case r.Kind == rowHint:
		return ui.Dim(r.Text)
	default:
		return r.Text
	}
}

// viewTitle 各视图标题行
func viewTitle(m *Model) string {
	switch m.View {
	case viewGlobal:
		done, total := 0, 0
		for bi := range m.RM.Branches {
			d, t := m.progress(bi)
			done, total = done+d, total+t
		}
		pct := 0
		if total > 0 {
			pct = done * 100 / total
		}
		return fmt.Sprintf("\x1b[1mGo 学习路线图\x1b[22m　%d/%d (%d%%)", done, total, pct)
	case viewBranch:
		b := &m.RM.Branches[m.Branch]
		done, total := m.progress(m.Branch)
		return fmt.Sprintf("\x1b[1m%s\x1b[22m　%d/%d", b.Title, done, total)
	default:
		p := m.curPoint()
		return fmt.Sprintf("\x1b[1m%s\x1b[22m　%s　%s", p.Title, roadmap.Glyph(p.Status), m.RM.PathOf(roadmap.PointRef{m.Branch, m.Point}))
	}
}

// viewHint 各视图键位提示
func viewHint(m *Model) string {
	switch m.View {
	case viewGlobal:
		return ui.Dim("↑↓/jk 移动　Tab 展开/收起　Enter 进入　d/s/t 标记　Esc/q 退出")
	case viewBranch:
		return ui.Dim("↑↓/jk 移动　Enter 看笔记　d/s/t 标记　Esc 返回　q 退出")
	default:
		return ui.Dim("↑↓ 选笔记　Enter/o 打开　d/s/t 标记状态　Esc 返回　q 退出")
	}

// drawFrame 重绘整帧：标题 + 键位提示 + 视口行（光标反白）+ 错误行；
// 写完光标移回帧首，供下次整体重绘覆盖
func drawFrame(out io.Writer, m *Model) {
	rows := m.Rows()
	avail := 0
	if h, ok := ui.PageSize(); ok {
		avail = h - 5
	}
	lo, hi := window(len(rows), m.Cursor, avail)
	width := ui.TermWidth(90)
	if width > 100 {
		width = 100
	}
	var lines []string
	if lo > 0 {
		lines = append(lines, ui.Dim(fmt.Sprintf("　…（上方还有 %d 行）", lo)))
	}
	for i := lo; i < hi; i++ {
		lines = append(lines, renderRow(rows[i], i == m.Cursor, width))
	}
	if hi < len(rows) {
		lines = append(lines, ui.Dim(fmt.Sprintf("　…（下方还有 %d 行）", len(rows)-hi)))
	}
	var b strings.Builder
	b.WriteString("\r\x1b[J")
	b.WriteString(viewTitle(m) + "\r\n")
	b.WriteString(viewHint(m) + "\r\n")
	for _, l := range lines {
		b.WriteString(l + "\r\n")
	}
	if m.Err != "" {
		b.WriteString(ui.Err(m.Err) + "\r\n")
	}
	fmt.Fprintf(&b, "\x1b[%dA\r", 1+len(lines))
	fmt.Fprint(out, b.String())
}

package planview

import (
	"errors"
	"fmt"
	"io"

	"github.com/bugzi/gs/internal/roadmap"
	"github.com/bugzi/gs/internal/termkey"
)

// ErrNonInteractive raw 模式不可用（非 TTY 等）；调用方降级静态输出
var ErrNonInteractive = errors.New("非交互终端，降级静态输出")

// Hooks 浏览器对外回调：打开笔记（壳层退 raw 后调 pager）与状态落盘
type Hooks struct {
	OpenNote func(rel string)
	Save     func(rm *roadmap.Roadmap) error
}

// keySession raw 按键会话：暂停（让给 pager）后可重开，退出保证还原终端
type keySession struct{ kr *termkey.KeyReader }

func (s *keySession) resume() bool {
	if s.kr != nil {
		return true
	}
	kr, err := termkey.NewKeyReader()
	if err != nil {
		return false
	}
	s.kr = kr
	return true
}

func (s *keySession) close() {
	if s.kr != nil {
		s.kr.Close()
		s.kr = nil
	}
}

// Run 浏览器主循环；原地变更 rm。raw 不可用返回 ErrNonInteractive。
func Run(out io.Writer, rm *roadmap.Roadmap, root, focusBranch string, h Hooks) error {
	sess := &keySession{}
	defer sess.close()
	if !sess.resume() {
		return ErrNonInteractive
	}
	m := &Model{RM: rm, Root: root, View: viewGlobal, Expanded: map[int]bool{}}
	if focusBranch != "" {
		m.Focus(focusBranch)
	}
	for {
		m.Normalize()
		drawFrame(out, m)
		k, ok := sess.kr.Read()
		if !ok {
			break
		}
		m.Err = ""
		switch k.Kind {
		case termkey.KindUp:
			m.Move(-1)
		case termkey.KindDown:
			m.Move(1)
		case termkey.KindTab:
			m.Toggle()
		case termkey.KindEnter:
			act, rel := m.Enter()
			if act == ActOpenNote {
				sess.close()
				fmt.Fprint(out, "\r\x1b[J\r\n")
				if h.OpenNote != nil {
					h.OpenNote(rel)
				}
				if !sess.resume() {
					return ErrNonInteractive
				}
			}
		case termkey.KindEsc:
			if !m.Back() {
				sess.close()
				fmt.Fprint(out, "\r\x1b[J\r\n")
				return nil
			}
		case termkey.KindCtrlC, termkey.KindCtrlD:
			sess.close()
			fmt.Fprint(out, "\r\x1b[J\r\n")
			return nil
		case termkey.KindRune:
			switch k.R {
			case 'q':
				sess.close()
				fmt.Fprint(out, "\r\x1b[J\r\n")
				return nil
			case 'j':
				m.Move(1)
			case 'k':
				m.Move(-1)
			case 'o':
				if act, rel := m.Enter(); act == ActOpenNote {
					sess.close()
					fmt.Fprint(out, "\r\x1b[J\r\n")
					if h.OpenNote != nil {
						h.OpenNote(rel)
					}
					if !sess.resume() {
						return ErrNonInteractive
					}
				}
			case 'd', 's', 't':
				status := roadmap.StatusDone
				if k.R == 's' {
					status = roadmap.StatusDoing
				} else if k.R == 't' {
					status = roadmap.StatusTodo
				}
				if _, ok := m.Mark(status); ok && h.Save != nil {
					if err := h.Save(m.RM); err != nil {
						m.Err = "落盘失败：" + err.Error()
					}
				}
			}
		}
	}
	return nil
}

// loadRoadmapSoft 路线图不存在时返回 nil 不报错，供可选挂载路径
func loadRoadmapSoft(ctx *Ctx) (*roadmap.Roadmap, error) {
	if _, err := os.Stat(ctx.WS.RoadmapFile()); err != nil {
		return nil, nil
	}
	return roadmap.Load(ctx.WS.RoadmapFile())
}

// locatePoint 知识点定位：关键词唯一直达、多匹配弹选、空关键词全量弹选
func locatePoint(ctx *Ctx, rm *roadmap.Roadmap, kw string) (roadmap.PointRef, error) {
	matches := rm.MatchPoints(kw)
	switch len(matches) {
	case 0:
		if kw == "" {
			return roadmap.PointRef{}, errors.New("路线图没有任何知识点，可用 /plan add point 添加")
		}
		return roadmap.PointRef{}, fmt.Errorf("没有匹配 %q 的知识点", kw)
	case 1:
		return matches[0], nil
	}
	var pick string
	opts := make([]huhOption, 0, len(matches))
	for _, m := range matches {
		opts = append(opts, option(rm.PathOf(m)+"（当前 "+rm.PointAt(m).Status+"）", pointKey(m)))
	}
	if err := selectFrom(ctx, "选择知识点", opts, &pick); err != nil {
		return roadmap.PointRef{}, err
	}
	return pointParse(pick), nil
}

func pointKey(r roadmap.PointRef) string { return fmt.Sprintf("%d/%d", r.Branch, r.Point) }

func pointParse(s string) roadmap.PointRef {
	var r roadmap.PointRef
	fmt.Sscanf(s, "%d/%d", &r.Branch, &r.Point)
	return r
}
