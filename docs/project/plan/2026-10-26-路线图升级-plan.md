**实现期决策（本计划锁定）：**

1. **树视图为纯文本直接输出**（不经 ui.RenderMD——markdown 渲染会破坏 ANSI 上色）；`/plan status` 仍走 RenderMD 不变。
2. **配色**：doing=220 金 / done=144 沙 / todo=245 灰（ui 现有低饱和暖色系同族，符合硬性规则 4）；roadmap 包经 `ColorFn(status, line) string` 回调注入，壳层用 lipgloss 上色。
3. `roadmap.Save` 重编码会丢弃文件内手写注释——「命令维护为主、手编兜底」定位下属预期。
4. **空主干**（如 I 算法与面试）树行不挂进度。
5. dot 引导线按 go-runewidth 显示宽对齐（间接依赖转直接，`go mod tidy`）。
6. roadmap.toml 初稿 80 叶全 todo（status 缺省即 todo，初稿省略该字段），调研标注高频的叶子打 `tags = ["高频"]`；F/G/H 只建枝干，I 只挂主干。
7. 多匹配选择器 value 用 `refKey`（`b/t/n` 下标串）传递，选中后 `refParse` 还原。
8. 状态符号统一 `roadmap.Glyph`：done `[x]` / doing `[~]` / todo `[ ]`。

// Package roadmap Go 学习路线图的纯逻辑：TOML 正本解析/写回、关键词匹配、状态变更、树渲染。
// 本包不 import ui/lipgloss，颜色由调用方注入。
package roadmap

import (
	"bytes"
	"errors"
	"fmt"
	"os"

	"github.com/BurntSushi/toml"

	"github.com/bugzi/gs/internal/workspace"
)

const (
	// Version 路线图文件格式版本
	Version = 1

	StatusTodo  = "todo"
	StatusDoing = "doing"
	StatusDone  = "done"
)

// Node 叶子知识点；Links 为挂载的知识笔记（工作区相对路径，知识绑定轮启用）
type Node struct {
	Title  string   `toml:"title"`
	Status string   `toml:"status"` // todo(缺省) | doing | done
	Tags   []string `toml:"tags,omitempty"`
	Links  []string `toml:"links,omitempty"`
	Note   string   `toml:"note,omitempty"`
}

// Topic 枝干
type Topic struct {
	Title string `toml:"title"`
	Nodes []Node `toml:"node,omitempty"`
}

// Branch 主干
type Branch struct {
	Title  string  `toml:"title"`
	Topics []Topic `toml:"topic,omitempty"`
}

// Roadmap 树状路线图正本
type Roadmap struct {
	Version  int      `toml:"version"`
	Branches []Branch `toml:"branch"`
}

// Load 读入并校验 roadmap.toml
func Load(path string) (*Roadmap, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	return Parse(data)
}

// Parse 解析；TOML 语法错误带行列号返回
func Parse(data []byte) (*Roadmap, error) {
	var rm Roadmap
	if err := toml.Unmarshal(data, &rm); err != nil {
		var perr toml.ParseError
		if errors.As(err, &perr) {
			return nil, fmt.Errorf("roadmap.toml 解析失败：\n%s", perr.ErrorWithPosition())
		}
		return nil, err
	}
	if err := rm.Validate(); err != nil {
		return nil, err
	}
	return &rm, nil
}

// Validate 结构校验：版本、主干非空、标题非空、status 合法（缺省归一化为 todo）
func (rm *Roadmap) Validate() error {
	if rm.Version != Version {
		return fmt.Errorf("不支持的版本 %d（期望 %d）", rm.Version, Version)
	}
	if len(rm.Branches) == 0 {
		return errors.New("路线图没有任何主干")
	}
	for bi := range rm.Branches {
		b := &rm.Branches[bi]
		if b.Title == "" {
			return fmt.Errorf("第 %d 个主干缺少标题", bi+1)
		}
		for ti := range b.Topics {
			tp := &b.Topics[ti]
			if tp.Title == "" {
				return fmt.Errorf("主干 %q 第 %d 个枝干缺少标题", b.Title, ti+1)
			}
			for ni := range tp.Nodes {
				if err := tp.Nodes[ni].normalize(); err != nil {
					return fmt.Errorf("主干 %q › 枝干 %q：%v", b.Title, tp.Title, err)
				}
			}
		}
	}
	return nil
}

func (n *Node) normalize() error {
	if n.Title == "" {
		return errors.New("知识点缺少标题")
	}
	switch n.Status {
	case "":
		n.Status = StatusTodo
	case StatusTodo, StatusDoing, StatusDone:
	default:
		return fmt.Errorf("知识点 %q 状态非法 %q（todo/doing/done）", n.Title, n.Status)
	}
	return nil
}

// Save 原子写回（重编码会丢弃手写注释，命令维护为主属预期）
func Save(path string, rm *Roadmap) error {
	var buf bytes.Buffer
	if err := toml.NewEncoder(&buf).Encode(rm); err != nil {
		return err
	}
	return workspace.WriteFileAtomic(path, buf.Bytes())
}

package roadmap

import (
	"errors"
	"fmt"
	"strings"
)

// NodeRef 叶子定位：主干/枝干/叶子三段下标
type NodeRef struct{ Branch, Topic, Node int }

// NodeAt 取叶子指针
func (rm *Roadmap) NodeAt(ref NodeRef) *Node {
	return &rm.Branches[ref.Branch].Topics[ref.Topic].Nodes[ref.Node]
}

// PathOf 叶子全路径「主干 › 枝干 › 叶子」
func (rm *Roadmap) PathOf(ref NodeRef) string {
	return rm.Branches[ref.Branch].Title + " › " +
		rm.Branches[ref.Branch].Topics[ref.Topic].Title + " › " +
		rm.NodeAt(ref).Title
}

// MatchNodes 关键词匹配全树叶子标题（大小写不敏感的包含匹配），按树序返回
func (rm *Roadmap) MatchNodes(kw string) []NodeRef {
	kw = strings.ToLower(kw)
	var out []NodeRef
	for bi := range rm.Branches {
		for ti := range rm.Branches[bi].Topics {
			for ni := range rm.Branches[bi].Topics[ti].Nodes {
				if strings.Contains(strings.ToLower(rm.Branches[bi].Topics[ti].Nodes[ni].Title), kw) {
					out = append(out, NodeRef{bi, ti, ni})
				}
			}
		}
	}
	return out
}

// MatchBranches 主干标题含关键词的下标
func (rm *Roadmap) MatchBranches(kw string) []int {
	kw = strings.ToLower(kw)
	var out []int
	for i := range rm.Branches {
		if strings.Contains(strings.ToLower(rm.Branches[i].Title), kw) {
			out = append(out, i)
		}
	}
	return out
}

// MatchTopics 某主干内枝干标题含关键词的下标
func (rm *Roadmap) MatchTopics(bi int, kw string) []int {
	kw = strings.ToLower(kw)
	var out []int
	for i := range rm.Branches[bi].Topics {
		if strings.Contains(strings.ToLower(rm.Branches[bi].Topics[i].Title), kw) {
			out = append(out, i)
		}
	}
	return out
}

// SetStatus 标记叶子状态
func (rm *Roadmap) SetStatus(ref NodeRef, status string) error {
	switch status {
	case StatusTodo, StatusDoing, StatusDone:
	default:
		return fmt.Errorf("非法状态 %q", status)
	}
	rm.NodeAt(ref).Status = status
	return nil
}

// TotalNodes 全树叶子数
func (rm *Roadmap) TotalNodes() int {
	n := 0
	for bi := range rm.Branches {
		for ti := range rm.Branches[bi].Topics {
			n += len(rm.Branches[bi].Topics[ti].Nodes)
		}
	}
	return n
}

// DoneNodes 全树 done 叶子数
func (rm *Roadmap) DoneNodes() int {
	n := 0
	for bi := range rm.Branches {
		for ti := range rm.Branches[bi].Topics {
			for ni := range rm.Branches[bi].Topics[ti].Nodes {
				if rm.Branches[bi].Topics[ti].Nodes[ni].Status == StatusDone {
					n++
				}
			}
		}
	}
	return n
}

// BranchProgress 主干内 done/total
func (rm *Roadmap) BranchProgress(bi int) (done, total int) {
	for ti := range rm.Branches[bi].Topics {
		for ni := range rm.Branches[bi].Topics[ti].Nodes {
			total++
			if rm.Branches[bi].Topics[ti].Nodes[ni].Status == StatusDone {
				done++
			}
		}
	}
	return done, total
}

// AddBranch 新主干，重名报错
func (rm *Roadmap) AddBranch(title string) error {
	if strings.TrimSpace(title) == "" {
		return errors.New("标题不能为空")
	}
	for i := range rm.Branches {
		if rm.Branches[i].Title == title {
			return fmt.Errorf("主干已存在: %s", title)
		}
	}
	rm.Branches = append(rm.Branches, Branch{Title: title})
	return nil
}

// AddTopic 主干下新枝干，同主干查重
func (rm *Roadmap) AddTopic(bi int, title string) error {
	if strings.TrimSpace(title) == "" {
		return errors.New("标题不能为空")
	}
	for i := range rm.Branches[bi].Topics {
		if rm.Branches[bi].Topics[i].Title == title {
			return fmt.Errorf("枝干已存在: %s", title)
		}
	}
	rm.Branches[bi].Topics = append(rm.Branches[bi].Topics, Topic{Title: title})
	return nil
}

// AddNode 枝干下新叶子（status=todo），同枝干查重，跨枝干允许重名
func (rm *Roadmap) AddNode(bi, ti int, title string, tags []string) error {
	if strings.TrimSpace(title) == "" {
		return errors.New("标题不能为空")
	}
	nodes := rm.Branches[bi].Topics[ti].Nodes
	for i := range nodes {
		if nodes[i].Title == title {
			return fmt.Errorf("知识点已存在: %s", title)
		}
	}
	rm.Branches[bi].Topics[ti].Nodes = append(nodes, Node{Title: title, Status: StatusTodo, Tags: tags})
	return nil
}

package roadmap

import (
	"fmt"
	"strings"

	"github.com/mattn/go-runewidth"

// ColorFn 状态行上色回调：返回包装后的行；nil 表示不上色。
// 由壳层注入（lipgloss），本包不 import ui，保持纯逻辑可独立测试。
type ColorFn func(status, line string) string

// RenderOpts 渲染选项
type RenderOpts struct {
	Title     string  // 树标题（如「Go 学习路线图」）
	Progress  string  // 树顶进度段（如「进度 23/96 (24%)」），空则不显示
	Next      string  // 树顶断点摘要行，空则不显示
	BranchIdx []int   // 只渲染这些主干；nil = 全部
	Color     ColorFn // nil = 不上色
}

// Glyph 状态符号：done [x] / doing [~] / 其他 [ ]
func Glyph(status string) string {
	switch status {
	case StatusDone:
		return "[x]"
	case StatusDoing:
		return "[~]"
	default:
		return "[ ]"
	}
}

// Render 渲染路线图树（纯函数）：主干行挂 done/total（空主干不挂），
// 「高频」尾随 *，主干过滤只输出指定主干。
func Render(rm *Roadmap, o RenderOpts) string {
	var b strings.Builder
	if o.Progress != "" {
		b.WriteString(leaderLine(o.Title, o.Progress, 44) + "\n")
	} else {
		b.WriteString(o.Title + "\n")
	}
	if o.Next != "" {
		b.WriteString(o.Next + "\n")
	}
	b.WriteString(strings.Repeat("═", 44) + "\n")

	ids := o.BranchIdx
	if ids == nil {
		ids = make([]int, len(rm.Branches))
		for i := range rm.Branches {
			ids[i] = i
		}
	}
	for k, bi := range ids {
		branch := &rm.Branches[bi]
		bg, bc := "├─ ", "│  "
		if k == len(ids)-1 {
			bg, bc = "└─ ", "   "
		}
		line := bg + branch.Title
		if done, total := rm.BranchProgress(bi); total > 0 {
			line = leaderLine(line, fmt.Sprintf("%d/%d", done, total), 40)
		}
		b.WriteString(line + "\n")
		for t := range branch.Topics {
			tp := &branch.Topics[t]
			tg, tc := "├─ ", "│  "
			if t == len(branch.Topics)-1 {
				tg, tc = "└─ ", "   "
			}
			b.WriteString(bc + tg + tp.Title + "\n")
			for n := range tp.Nodes {
				nd := &tp.Nodes[n]
				ng := "├─ "
				if n == len(tp.Nodes)-1 {
					ng = "└─ "
				}
				nl := bc + tc + ng + Glyph(nd.Status) + " " + nd.Title
				if hasHighFreq(nd.Tags) {
					nl += " *"
				}
				if o.Color != nil {
					nl = o.Color(nd.Status, nl)
				}
				b.WriteString(nl + "\n")
			}
		}
	}
	return b.String()
}

// leaderLine 「title ···· progress」三段行，目标总显示宽 w（中点至少 3 个）
func leaderLine(title, progress string, w int) string {
	dots := w - runewidth.StringWidth(title) - runewidth.StringWidth(progress) - 2
	if dots < 3 {
		dots = 3
	}
	return title + " " + strings.Repeat("·", dots) + " " + progress
}

// hasHighFreq tags 含「高频」即视为高频
func hasHighFreq(tags []string) bool {
	for _, tg := range tags {
		if strings.Contains(tg, "高频") {
			return true
		}
	}
	return false
}

		return nil, errors.New("路线图尚未生成（Study/plan/roadmap.toml 不存在），可先用 /plan status 查看学习状态")
	}

// runPlanTree 默认视图：全景树（或主干过滤）pager；e 直接编辑 roadmap.toml
func runPlanTree(ctx *Ctx, kw string) error {
	rm, err := loadRoadmap(ctx)
	if err != nil {
		return err
	}
	if kw != "" && len(rm.MatchBranches(kw)) == 0 {
		names := make([]string, 0, len(rm.Branches))
		for i := range rm.Branches {
			names = append(names, rm.Branches[i].Title)
		}
		return fmt.Errorf("没有匹配 %q 的主干；可用主干：%s", kw, strings.Join(names, "、"))
	}
	path := ctx.WS.RoadmapFile()
	return runViewPager(ctx,
		func() string {
			rm2, err := roadmap.Load(path)
			if err != nil {
				return ui.Err(err.Error())
			}
			var st *State
			if data, err := os.ReadFile(ctx.WS.StateFile()); err == nil {
				s := ParseState(string(data))
				st = &s
			}
			return planTreeText(rm2, st, kw)
		},
		func() (string, bool) { return path, true })
}

// stateNext 取学习状态「下一步」键值，无则空串
func stateNext(st State) string {
	for _, kv := range st.Breakpoints {
		if kv.Key == "下一步" {
			return kv.Value
		}
	}
	return ""
}

// planColor 状态行配色：doing 金 220 / done 沙 144 / todo 灰 245（同族低饱和）
func planColor(status, line string) string {
	c := "245"
	switch status {
	case roadmap.StatusDoing:
		c = "220"
	case roadmap.StatusDone:
		c = "144"
	}
	return lipgloss.NewStyle().Foreground(lipgloss.Color(c)).Render(line)
}

// planTreeText 组装树视图文本（渲染 + 断点摘要降级），纯逻辑可测
func planTreeText(rm *roadmap.Roadmap, st *State, kw string) string {
	next := "（未找到学习状态.md）"
	if st != nil {
		if v := stateNext(*st); v != "" {
			next = "下一步：" + v
		} else {
			next = "（学习状态.md 没有「下一步」断点）"
		}
	}
	done, total := rm.DoneNodes(), rm.TotalNodes()
	pct := 0
	if total > 0 {
		pct = done * 100 / total
	}
	var idx []int
	if kw != "" {
		idx = rm.MatchBranches(kw)
	}
	return roadmap.Render(rm, roadmap.RenderOpts{
		Title:     "Go 学习路线图",
		Progress:  fmt.Sprintf("进度 %d/%d (%d%%)", done, total, pct),
		Next:      next,
		BranchIdx: idx,
		Color:     planColor,
	})
}

// refKey / refParse NodeRef 与选择器 value 的互转（"b/t/n" 下标串）
func refKey(r roadmap.NodeRef) string { return fmt.Sprintf("%d/%d/%d", r.Branch, r.Topic, r.Node) }

func refParse(s string) roadmap.NodeRef {
	var r roadmap.NodeRef
	fmt.Sscanf(s, "%d/%d/%d", &r.Branch, &r.Topic, &r.Node)
	return r
}

// pickNodeRef 选叶子：有关键词走匹配三态，无参 主干→枝干→叶子 逐层选
func pickNodeRef(ctx *Ctx, rm *roadmap.Roadmap, kw, label string) (roadmap.NodeRef, error) {
	if kw != "" {
		matches := rm.MatchNodes(kw)
		switch len(matches) {
		case 0:
			return roadmap.NodeRef{}, fmt.Errorf("没有匹配 %q 的知识点", kw)
		case 1:
			return matches[0], nil
		}
		var pick string
		opts := make([]huhOption, 0, len(matches))
		for _, m := range matches {
			opts = append(opts, option(rm.PathOf(m)+"（当前 "+rm.NodeAt(m).Status+"）", refKey(m)))
		}
		if err := selectFrom(ctx, label+"（多匹配）", opts, &pick); err != nil {
			return roadmap.NodeRef{}, err
		}
		return refParse(pick), nil
	}
	bi, err := pickBranchIdx(ctx, rm)
	if err != nil {
		return roadmap.NodeRef{}, err
	}
	ti, err := pickTopicIdx(ctx, rm, bi)
	if err != nil {
		return roadmap.NodeRef{}, err
	}
	nodes := rm.Branches[bi].Topics[ti].Nodes
	if len(nodes) == 0 {
		return roadmap.NodeRef{}, fmt.Errorf("枝干 %q 下没有知识点，可用 /plan add node 添加", rm.Branches[bi].Topics[ti].Title)
	}
	var pick string
	opts := make([]huhOption, 0, len(nodes))
	for ni := range nodes {
		n := &nodes[ni]
		opts = append(opts, option(roadmap.Glyph(n.Status)+" "+n.Title, fmt.Sprintf("%d", ni)))
	}
	if err := selectFrom(ctx, "选择知识点", opts, &pick); err != nil {
		return roadmap.NodeRef{}, err
	}
	var ni int
	fmt.Sscanf(pick, "%d", &ni)
	return roadmap.NodeRef{bi, ti, ni}, nil
}

// pickBranchIdx 无参纯交互选主干
func pickBranchIdx(ctx *Ctx, rm *roadmap.Roadmap) (int, error) {
	all := make([]int, len(rm.Branches))
	for i := range rm.Branches {
		all[i] = i
	}
	return pickBranchFrom(ctx, rm, all, "选择主干")
}

// pickBranchFrom 在候选主干下标中弹选择
func pickBranchFrom(ctx *Ctx, rm *roadmap.Roadmap, ids []int, label string) (int, error) {
	var pick string
	opts := make([]huhOption, 0, len(ids))
	for _, i := range ids {
		done, total := rm.BranchProgress(i)
		opts = append(opts, option(fmt.Sprintf("%s（%d/%d）", rm.Branches[i].Title, done, total), fmt.Sprintf("%d", i)))
	}
	if err := selectFrom(ctx, label, opts, &pick); err != nil {
		return 0, err
	}
	var bi int
	fmt.Sscanf(pick, "%d", &bi)
	return bi, nil
}

// pickTopicIdx 无参纯交互选枝干
func pickTopicIdx(ctx *Ctx, rm *roadmap.Roadmap, bi int) (int, error) {
	tps := rm.Branches[bi].Topics
	if len(tps) == 0 {
		return 0, fmt.Errorf("主干 %q 下没有枝干，可用 /plan add topic 添加", rm.Branches[bi].Title)
	}
	var pick string
	opts := make([]huhOption, 0, len(tps))
	for i := range tps {
		opts = append(opts, option(fmt.Sprintf("%s（%d 个知识点）", tps[i].Title, len(tps[i].Nodes)), fmt.Sprintf("%d", i)))
	}
	if err := selectFrom(ctx, "选择枝干", opts, &pick); err != nil {
		return 0, err
	}
	var ti int
	fmt.Sscanf(pick, "%d", &ti)
	return ti, nil
}

// runPlanAdd 建树期录节点：plan add branch|topic|node，缺参交互补全
func runPlanAdd(ctx *Ctx, args []string) error {
	rm, err := loadRoadmap(ctx)
	if err != nil {
		return err
	}
	kind := ""
	if len(args) > 0 {
		kind = args[0]
		args = args[1:]
	}
	switch kind {
	case "branch":
		title, err := takeArg(ctx, args, "主干标题")
		if err != nil {
			return err
		}
		if err := rm.AddBranch(title); err != nil {
			return err
		}
	case "topic":
		bi, rest, err := pickBranchByArgs(ctx, rm, args)
		if err != nil {
			return err
		}
		title, err := takeArg(ctx, rest, "枝干标题")
		if err != nil {
			return err
		}
		if err := rm.AddTopic(bi, title); err != nil {
			return err
		}
	case "node":
		bi, rest, err := pickBranchByArgs(ctx, rm, args)
		if err != nil {
			return err
		}
		ti, rest2, err := pickTopicByArgs(ctx, rm, bi, rest)
		if err != nil {
			return err
		}
		title, err := takeArg(ctx, rest2, "知识点标题")
		if err != nil {
			return err
		}
		if err := rm.AddNode(bi, ti, title, nil); err != nil {
			return err
		}
	default:
		return fmt.Errorf("未知类型 %q（支持 branch / topic / node）", kind)
	}
	if err := roadmap.Save(ctx.WS.RoadmapFile(), rm); err != nil {
		return err
	}
	fmt.Fprintln(ctx.Out, "已添加并落盘 roadmap.toml")
	return nil
}

// takeArg 取剩余参数作标题；为空则交互输入
func takeArg(ctx *Ctx, args []string, label string) (string, error) {
	if s := strings.TrimSpace(strings.Join(args, " ")); s != "" {
		return s, nil
	}
	return inputLine(ctx, label)
}

// pickBranchByArgs 首参作主干关键词定位（唯一直达/多匹配选/零匹配报错）；无参交互选。
// 返回选中的主干下标与剩余参数。
func pickBranchByArgs(ctx *Ctx, rm *roadmap.Roadmap, args []string) (int, []string, error) {
	if len(args) == 0 {
		bi, err := pickBranchIdx(ctx, rm)
		return bi, nil, err
	}
	ms := rm.MatchBranches(args[0])
	switch len(ms) {
	case 0:
		return 0, nil, fmt.Errorf("没有匹配 %q 的主干", args[0])
	case 1:
		return ms[0], args[1:], nil
	}
	bi, err := pickBranchFrom(ctx, rm, ms, "匹配到多个主干")
	if err != nil {
		return 0, nil, err
	}
	return bi, args[1:], nil
}

// pickTopicByArgs 首参作枝干关键词定位；无参交互选。返回枝干下标与剩余参数。
func pickTopicByArgs(ctx *Ctx, rm *roadmap.Roadmap, bi int, args []string) (int, []string, error) {
	if len(args) == 0 {
		ti, err := pickTopicIdx(ctx, rm, bi)
		return ti, nil, err
	}
	ms := rm.MatchTopics(bi, args[0])
	switch len(ms) {
	case 0:
		return 0, nil, fmt.Errorf("没有匹配 %q 的枝干", args[0])
	case 1:
		return ms[0], args[1:], nil
	}
	var pick string
	opts := make([]huhOption, 0, len(ms))
	for _, i := range ms {
		opts = append(opts, option(rm.Branches[bi].Topics[i].Title, fmt.Sprintf("%d", i)))
	}
	if err := selectFrom(ctx, "匹配到多个枝干", opts, &pick); err != nil {
		return 0, nil, err
	}
	var ti int
	fmt.Sscanf(pick, "%d", &ti)
	return ti, args[1:], nil
}
