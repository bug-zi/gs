// CollectTargets 递归收集应生成介绍的节点：全部目录（不含根）+ 全部 .go 文件。
// 跳过隐藏目录与 exclude 段名，与树扫描规则一致；按 Rel 排序返回稳定列表。
func CollectTargets(projectPath string, exclude []string) []Entry {
	ex := map[string]bool{}
	for _, e := range exclude {
		ex[e] = true
	}
	var out []Entry
	filepath.WalkDir(projectPath, func(p string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil
		}
		name := d.Name()
		if d.IsDir() {
			if p == projectPath {
				return nil // 根目录本身不是树节点
			}
			if strings.HasPrefix(name, ".") || ex[name] {
				return filepath.SkipDir
			}
			rel, err := filepath.Rel(projectPath, p)
			if err != nil {
				return nil
			}
			out = append(out, Entry{Name: name, Rel: filepath.ToSlash(rel), IsDir: true})
			return nil
		}
		if !strings.EqualFold(filepath.Ext(name), ".go") {
			return nil
		}
		rel, err := filepath.Rel(projectPath, p)
		if err != nil {
			return nil
		}
		out = append(out, Entry{Name: name, Rel: filepath.ToSlash(rel), IsDir: false})
		return nil
	})
	sort.Slice(out, func(i, j int) bool { return out[i].Rel < out[j].Rel })
	return out
}
```

// Missing 返回 targets 中缓存尚无介绍的节点（保持入参顺序）
func (c *StructureCache) Missing(targets []Entry) []Entry {
	var out []Entry
	for _, t := range targets {
		if c.Find(t.Rel) == "" {
			out = append(out, t)
		}
	}
	return out
}

// ClearSubtree 删除 prefix 子树的全部缓存条目（含 prefix 自身）；prefix 为空清空全部
func (c *StructureCache) ClearSubtree(prefix string) {
	kept := c.Nodes[:0]
	for _, n := range c.Nodes {
		if prefix == "" || n.Path == prefix || strings.HasPrefix(n.Path, prefix+"/") {
			continue
		}
		kept = append(kept, n)
	}
	c.Nodes = kept
}
```

// filterSubtree 只保留 dir 子树内（含 dir 自身）的节点
func filterSubtree(entries []structure.Entry, dir string) []structure.Entry {
	prefix := dir + "/"
	kept := entries[:0]
	for _, e := range entries {
		if e.Rel == dir || strings.HasPrefix(e.Rel, prefix) {
			kept = append(kept, e)
		}
	}
	return kept
}

// splitBatches 缺失节点按父目录分组；返回自上而下排序的父目录列表与分组映射
// （字典序即自上而下：父路径是子路径前缀，必排在前）
func splitBatches(missing []structure.Entry) ([]string, map[string][]structure.Entry) {
	byDir := map[string][]structure.Entry{}
	for _, t := range missing {
		parent := parentDir(t.Rel)
		byDir[parent] = append(byDir[parent], t)
	}
	dirs := make([]string, 0, len(byDir))
	for d := range byDir {
		dirs = append(dirs, d)
	}
	sort.Strings(dirs)
	return dirs, byDir
}
```

// genTreeDescriptions 生成 dir 子树内缺失节点的介绍：按目录分批，只询问缓存缺失者，
// 单批失败记录后继续其余批次，每批成功即并入缓存保存；返回失败批次目录列表。
func genTreeDescriptions(ctx *Ctx, p workspace.Project, cache *structure.StructureCache, dir string) ([]string, error) {
	st, err := ensureAIState(ctx)
	if err != nil {
		return nil, err
	}
	missing := cache.Missing(structure.CollectTargets(p.Path, ctx.Cfg.Scan.Exclude))
	if dir != "" {
		missing = filterSubtree(missing, dir)
	}
	if len(missing) == 0 {
		return nil, nil
	}
	dirs, byDir := splitBatches(missing)
	var failed []string
	for _, d := range dirs {
		label := d
		if label == "" {
			label = "（根目录）"
		}
		fmt.Fprintln(ctx.Out, ui.Dim("（AI 分析中… "+label+"）"))
		nodes, err := genLevelDescriptions(ctx, st, p, d, byDir[d])
		if err != nil {
			failed = append(failed, d)
			continue
		}
		cache.Merge(nodes)
		if err := cache.Save(structure.StructurePath(ctx.WS, p.Name)); err != nil {
			return failed, err
		}
	}
	return failed, nil
}
```
